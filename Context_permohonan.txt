  function formatJabatanSingkat(jab) {
      if (!jab) return "-";
      let j = jab.toUpperCase();
      if (j.includes("KEPALA SUB BAGIAN PERENCANAAN, TI DAN PELAPORAN")) return "Kasubbag PTIP";
      if (j.includes("KEPALA SUB BAGIAN UMUM DAN KEUANGAN")) return "Kasubbag Umum dan Keuangan";
      if (j.includes("KEPALA SUB BAGIAN KEPEGAWAIAN, ORTALA") || j.includes("KEPALA SUB BAGIAN KEPEGAWAIAN DAN ORTALA")) return "Kasubbag Kepegawaian & ORTALA";
      return jab;
  }

  function formatDateIndoCetak(dateStr) {
      if (!dateStr) return "-";
      const months = ["Januari", "Februari", "Maret", "April", "Mei", "Juni", "Juli", "Agustus", "September", "Oktober", "November", "Desember"];
      
      // Jika formatnya YYYY-MM-DD (contoh: dari kalender mulai/selesai cuti)
      let parts = dateStr.split('-');
      if (parts.length === 3) {
          return parseInt(parts[2], 10) + " " + months[parseInt(parts[1], 10) - 1] + " " + parts[0];
      }
      
      // Jika formatnya DD/MM/YYYY (contoh: dari database tgl_pengajuan)
      parts = dateStr.split('/');
      if (parts.length === 3) {
          return parseInt(parts[0], 10) + " " + months[parseInt(parts[1], 10) - 1] + " " + parts[2];
      }
      
      return dateStr;
  }

  function switchCutiTab(tabId) {
      document.querySelectorAll('.cuti-tab-content').forEach(function(el) {

          el.classList.remove('block'); 
          el.style.opacity = '0';
          el.style.transition = 'none';
      });
      
      const targetContent = document.getElementById('tab-' + tabId);
      if (targetContent) {
          targetContent.classList.remove('hidden');
          targetContent.classList.add('block');
          
          setTimeout(() => {
              targetContent.style.transition = 'opacity 0.4s ease-in-out';
              targetContent.style.opacity = '1';
          }, 20);
      }

      tabs.forEach(function(id) {
          const btn = document.getElementById('btn-tab-' + id);
          if (!btn) return;
          
          if (id === tabId) {
              btn.classList.add('bg-indigo-600', 'text-white', 'shadow-lg', 'shadow-indigo-600/20');
              btn.classList.remove('text-slate-400', 'hover:text-slate-200', 'hover:bg-slate-800');

                          else if (isAtasan && log.status_atasan === 'Menunggu') pendingCount++;

  function calculateDuration(start, end) {
      const d1 = new Date(start); const d2 = new Date(end);
      let count = 0;
      for (let d = new Date(d1); d <= d2; d.setDate(d.getDate() + 1)) {
          if (d.getDay() !== 0 && d.getDay() !== 6) count++;
      }
      return count;
  }

  function isPegawaiAbsent(nip) {
      const today = new Date();
      today.setHours(0,0,0,0);
      return state.cutiLogs.some(log => {
          if (String(log.nip) === String(nip) && log.status === 'Disetujui') {
              const start = new Date(log.tgl_mulai);
              start.setHours(0,0,0,0);
              const end = new Date(log.tgl_selesai);
              end.setHours(0,0,0,0);
              return today >= start && today <= end;
          }
          return false;
      });
  }

  function setupAtasanAndPejabat(targetUserObj) {
      const u = targetUserObj || state.user; 
      const j = u.jabatan.toLowerCase();
      let atasanOpts = [];
      let pejabatOpts = [];
      let adaPlh = false;

      function getPegawai(keyword) {
          return state.usersList.filter(x => x.nip !== u.nip && x.jabatan.toLowerCase().includes(keyword));
      }

      let ketua = getPegawai("ketua").find(x => x.jabatan.toLowerCase() === "ketua");
      if (ketua && !isPegawaiAbsent(ketua.nip)) {
          pejabatOpts.push(ketua);
      } else {
          adaPlh = true;
          let wakil = getPegawai("wakil ketua").find(x => x.jabatan.toLowerCase() === "wakil ketua");
          if (wakil && !isPegawaiAbsent(wakil.nip)) pejabatOpts.push(wakil); 
          else { let hakims = getPegawai("hakim").filter(x => !isPegawaiAbsent(x.nip)); pejabatOpts.push(...hakims); }
      }

      if (j === "ketua") { atasanOpts = state.usersList.filter(x => x.nip !== u.nip); }
      else if (j.includes("wakil ketua") || j.includes("hakim") || (j.includes("panitera") && !j.includes("muda") && !j.includes("pengganti")) || j === "sekretaris") {
          atasanOpts = [...pejabatOpts]; 
      }
      else if (j.includes("panitera muda") || j.includes("panitera pengganti") || j.includes("jurusita") || j.includes("juru sita")) {
          let panitera = getPegawai("panitera").find(x => x.jabatan.toLowerCase() === "panitera");
          if (panitera && !isPegawaiAbsent(panitera.nip)) { atasanOpts.push(panitera); } 
          else {
              adaPlh = true; let panmuds = getPegawai("panitera muda").filter(x => !isPegawaiAbsent(x.nip));
              atasanOpts.push(...panmuds);
          }
      }
      else if (j.includes("kepala sub bagian") || j.includes("kasubbag")) {
          let sekretaris = getPegawai("sekretaris").find(x => x.jabatan.toLowerCase() === "sekretaris");
          if (sekretaris && !isPegawaiAbsent(sekretaris.nip)) { atasanOpts.push(sekretaris); } 
          else {
              adaPlh = true; let kasubbags = getPegawai("kepala sub bagian").filter(x => !isPegawaiAbsent(x.nip));
              if(kasubbags.length === 0) kasubbags = getPegawai("kasubbag").filter(x => !isPegawaiAbsent(x.nip));
              atasanOpts.push(...kasubbags);
          }
      }
      else {
          let panmuds = getPegawai("panitera muda").filter(x => !isPegawaiAbsent(x.nip));
          let kasubbags = getPegawai("kepala sub bagian").filter(x => !isPegawaiAbsent(x.nip));
          if(kasubbags.length === 0) kasubbags = getPegawai("kasubbag").filter(x => !isPegawaiAbsent(x.nip));
          atasanOpts.push(...panmuds); atasanOpts.push(...kasubbags);
          let panitera = getPegawai("panitera").find(x => x.jabatan.toLowerCase() === "panitera");
          if (panitera && !isPegawaiAbsent(panitera.nip)) atasanOpts.push(panitera);
          let sekretaris = getPegawai("sekretaris").find(x => x.jabatan.toLowerCase() === "sekretaris");
          if (sekretaris && !isPegawaiAbsent(sekretaris.nip)) atasanOpts.push(sekretaris);
      }

      atasanOpts = [...new Map(atasanOpts.map(item => [item.nip, item])).values()];
      pejabatOpts = [...new Map(pejabatOpts.map(item => [item.nip, item])).values()];

      const selAtasan = document.getElementById('cutiAtasan');
      const selPejabat = document.getElementById('cutiPejabat');
      const alertPlh = document.getElementById('alertPlh');
      
      const selPlhCuti = document.getElementById('cutiPlh');
      const containerPlhCuti = document.getElementById('containerPlhCuti');
      if (selPlhCuti) selPlhCuti.innerHTML = '<option value="">-- Pilih Pejabat Plh --</option>';
      
      let opsiPlhCuti = [];
      let isPanitera = j === "panitera";
      let isSekretaris = j === "sekretaris";

      if (isPanitera) {
           opsiPlhCuti = getPegawai("panitera muda").filter(x => !isPegawaiAbsent(x.nip));
           if(containerPlhCuti) containerPlhCuti.classList.remove('hidden');
      } else if (isSekretaris) {
           opsiPlhCuti = getPegawai("kepala sub bagian").filter(x => !isPegawaiAbsent(x.nip));
           if(opsiPlhCuti.length === 0) opsiPlhCuti = getPegawai("kasubbag").filter(x => !isPegawaiAbsent(x.nip));
           if(containerPlhCuti) containerPlhCuti.classList.remove('hidden');
      } else {
           if(containerPlhCuti) containerPlhCuti.classList.add('hidden');
      }

      if (selPlhCuti) {
           opsiPlhCuti.forEach(o => { selPlhCuti.innerHTML += `<option value="${o.nip}">${o.nama} (${o.jabatan})</option>`; });
      }
      
      selAtasan.innerHTML = '<option value="">-- Pilih Atasan --</option>';
      atasanOpts.forEach(o => { selAtasan.innerHTML += `<option value="${o.nip}">${o.nama} (${o.jabatan})</option>`; });

      selPejabat.innerHTML = '<option value="">-- Pilih Pejabat Berwenang --</option>';
      pejabatOpts.forEach(o => { selPejabat.innerHTML += `<option value="${o.nip}">${o.nama} (${o.jabatan})</option>`; });
      
      if(atasanOpts.length === 1) selAtasan.value = atasanOpts[0].nip;
      if(pejabatOpts.length === 1) selPejabat.value = pejabatOpts[0].nip;
      if (adaPlh) alertPlh.classList.remove('hidden'); else alertPlh.classList.add('hidden');
  }

      }
      
      isBypassMode = isBypass;

      if (isBypass) {

  function readFileAsBase64(file) {
      return new Promise(function(resolve, reject) {
          const reader = new FileReader();
          reader.onload = function() { resolve(reader.result); };
          reader.onerror = function(error) { reject(error); };
          reader.readAsDataURL(file);
      });
  }

  async function submitCuti(e) {
    e.preventDefault();
    if(!state.user) { showNotify("Error Sesi", "Harap login ulang", "error"); return; }
    
    const fileInput = document.getElementById('cutiBukti');
    let fileData = null, fileName = null, fileMimeType = null;
    
    if (document.getElementById('cutiJenis').value !== 'Tahunan') {
        if (fileInput.files.length === 0) { showNotify("Validasi", "Bukti dukung wajib dilampirkan!", "error"); return; }
        const file = fileInput.files[0];
        if (file.size > 2 * 1024 * 1024) { showNotify("File Besar", "Maksimal 2MB.", "error"); return; }
        
        showLoading("Membaca dokumen lampiran...");
        try {
           const base64Str = await readFileAsBase64(file);
           fileData = base64Str.split(',')[1];
           fileName = file.name;
           fileMimeType = file.type;
        } catch(err) {
           hideLoading(); showNotify("Error", "Gagal memproses file.", "error"); return;
        }
    }

    let finalNip = state.user.nip;
    if (isBypassMode) {
        finalNip = document.getElementById('cutiBypassNip').value;
        if (!finalNip) { hideLoading(); showNotify("Validasi", "Pilih pegawai pemohon terlebih dahulu!", "error"); return; }
    }

    const payload = {
        nip: finalNip,
        jenis: document.getElementById('cutiJenis').value,
        tgl_mulai: document.getElementById('cutiMulai').value,
        tgl_selesai: document.getElementById('cutiSelesai').value,
        alasan: document.getElementById('cutiAlasan').value,
        alamat: document.getElementById('cutiAlamat').value,
        atasan_nip: document.getElementById('cutiAtasan').value,
        pejabat_nip: document.getElementById('cutiPejabat').value,
        plh_nip: document.getElementById('containerPlhCuti') && !document.getElementById('containerPlhCuti').classList.contains('hidden') ? document.getElementById('cutiPlh').value : "",
        fileData: fileData, fileName: fileName, fileMimeType: fileMimeType
    };

    if (new Date(payload.tgl_selesai) < new Date(payload.tgl_mulai)) {
       hideLoading();
       showNotify("Tanggal Invalid", "Selesai pelaksanaan tidak boleh sebelum tanggal mulai.", "error");
       return;
    }

    // Deteksi Tanggal Ganda (Overlapping Dates Conflict)
    const newStart = new Date(payload.tgl_mulai);
    newStart.setHours(0,0,0,0);
    const newEnd = new Date(payload.tgl_selesai);
    newEnd.setHours(0,0,0,0);

    const isOverlap = state.cutiLogs.some(log => {
        if (String(log.nip) === String(payload.nip) && log.status !== 'Ditolak' && log.status !== 'DIBATALKAN') {
            const existingStart = new Date(log.tgl_mulai);
            existingStart.setHours(0,0,0,0);
            const existingEnd = new Date(log.tgl_selesai);
            existingEnd.setHours(0,0,0,0);
            return (newStart <= existingEnd && newEnd >= existingStart);
        }
        return false;
    });

    if (isOverlap) {
        hideLoading();
        showNotify("Tanggal Bentrok", "Pengajuan gagal! Anda (atau pegawai ini) sudah memiliki izin/cuti yang aktif pada rentang tanggal tersebut.", "warning");
        return;
    }

    showLoading("Memverifikasi ketersediaan Pejabat...");
    
    google.script.run
        .withSuccessHandler(function(routeRes) {
            if (routeRes && routeRes.status === 'success') {
                payload.atasan_nip = routeRes.data.atasan_nip;
                payload.pejabat_nip = routeRes.data.pejabat_nip;

          if (String(log.nip) === String(nip) && log.status === 'Disetujui') {
              const start = new Date(log.tgl_mulai);
              start.setHours(0,0,0,0);
              const end = new Date(log.tgl_selesai);
              end.setHours(0,0,0,0);

  function readFileAsBase64(file) {
      return new Promise(function(resolve, reject) {
          const reader = new FileReader();
          reader.onload = function() { resolve(reader.result); };
          reader.onerror = function(error) { reject(error); };
          reader.readAsDataURL(file);
      });
  }

  async function submitCuti(e) {
    e.preventDefault();
    if(!state.user) { showNotify("Error Sesi", "Harap login ulang", "error"); return; }
    
    const fileInput = document.getElementById('cutiBukti');
    let fileData = null, fileName = null, fileMimeType = null;
    
    if (document.getElementById('cutiJenis').value !== 'Tahunan') {
        if (fileInput.files.length === 0) { showNotify("Validasi", "Bukti dukung wajib dilampirkan!", "error"); return; }
        const file = fileInput.files[0];
        if (file.size > 2 * 1024 * 1024) { showNotify("File Besar", "Maksimal 2MB.", "error"); return; }
        
        showLoading("Membaca dokumen lampiran...");
        try {
           const base64Str = await readFileAsBase64(file);
           fileData = base64Str.split(',')[1];
           fileName = file.name;
           fileMimeType = file.type;
        } catch(err) {
           hideLoading(); showNotify("Error", "Gagal memproses file.", "error"); return;
        }
    }

    let finalNip = state.user.nip;
    if (isBypassMode) {
        finalNip = document.getElementById('cutiBypassNip').value;
        if (!finalNip) { hideLoading(); showNotify("Validasi", "Pilih pegawai pemohon terlebih dahulu!", "error"); return; }
    }

    const payload = {
        nip: finalNip,
        jenis: document.getElementById('cutiJenis').value,
        tgl_mulai: document.getElementById('cutiMulai').value,
        tgl_selesai: document.getElementById('cutiSelesai').value,
        alasan: document.getElementById('cutiAlasan').value,
        alamat: document.getElementById('cutiAlamat').value,
        atasan_nip: document.getElementById('cutiAtasan').value,
        pejabat_nip: document.getElementById('cutiPejabat').value,
        plh_nip: document.getElementById('containerPlhCuti') && !document.getElementById('containerPlhCuti').classList.contains('hidden') ? document.getElementById('cutiPlh').value : "",
        fileData: fileData, fileName: fileName, fileMimeType: fileMimeType
    };

    if (new Date(payload.tgl_selesai) < new Date(payload.tgl_mulai)) {
       hideLoading();
       showNotify("Tanggal Invalid", "Selesai pelaksanaan tidak boleh sebelum tanggal mulai.", "error");
       return;
    }

    const newStart = new Date(payload.tgl_mulai);
    newStart.setHours(0,0,0,0);
    const newEnd = new Date(payload.tgl_selesai);
    newEnd.setHours(0,0,0,0);

    const isOverlap = state.cutiLogs.some(log => {
        if (String(log.nip) === String(payload.nip) && log.status !== 'Ditolak' && log.status !== 'DIBATALKAN') {
            const existingStart = new Date(log.tgl_mulai);
            existingStart.setHours(0,0,0,0);
            const existingEnd = new Date(log.tgl_selesai);
            existingEnd.setHours(0,0,0,0);
            return (newStart <= existingEnd && newEnd >= existingStart);
        }
        return false;
    });

    if (isOverlap) {
        hideLoading();
        showNotify("Tanggal Bentrok", "Pengajuan gagal! Anda (atau pegawai ini) sudah memiliki izin/cuti yang aktif pada rentang tanggal tersebut.", "warning");
        return;
    }

    // ZETTBOT FIX: Validasi Saldo Cuti di sisi Klien (Front-end)
    let d1 = new Date(payload.tgl_mulai);
    let d2 = new Date(payload.tgl_selesai);
    let durasi = 0;
    for (let d = new Date(d1); d <= d2; d.setDate(d.getDate() + 1)) {
        if (d.getDay() !== 0 && d.getDay() !== 6) durasi++;
    }

    let targetUser = state.user;
    if (isBypassMode) {
        targetUser = state.usersList.find(u => String(u.nip) === String(finalNip));
    }

    if (targetUser && payload.jenis !== 'Dinas Luar' && payload.jenis !== 'CLTN') {
        if (payload.jenis === 'Tahunan') {
            let totalTahunan = Number(targetUser.s_tahunan_2024 || 0) + Number(targetUser.s_tahunan_2025 || 0) + Number(targetUser.s_tahunan_2026 || 0);
            if (totalTahunan < durasi) {
                hideLoading();
                showNotify("Saldo Habis/Kurang", "Maaf, sisa kuota Cuti Tahunan Anda ("+totalTahunan+" Hari) tidak mencukupi untuk pengajuan durasi "+durasi+" Hari.", "error");
                return;
            }
        } else {
            let field = "";
            if(payload.jenis === 'Sakit') field = 's_sakit';
            else if(payload.jenis === 'Alasan Penting' || payload.jenis === 'Penting') field = 's_penting';

            else if(payload.jenis === 'Melahirkan') field = 's_melahirkan';

                if (sisa < durasi) {
                    hideLoading();
                    showNotify("Saldo Habis/Kurang", "Maaf, sisa kuota Cuti " + payload.jenis + " ("+sisa+" Hari) tidak mencukupi untuk pengajuan "+durasi+" Hari.", "error");
                    return;
                }
            }
        }
    }

    showLoading("Memverifikasi ketersediaan Pejabat...");
    
    google.script.run
        .withSuccessHandler(function(routeRes) {

                payload.atasan_nip = routeRes.data.atasan_nip;

                showLoading("Mengirimkan usulan cuti...");
                google.script.run
                  .withSuccessHandler(function(res) {

                    closeFormCutiModal();
                    if (res.status === 'success') {

                          showNotify("Berhasil Terkirim", res.message + "<br><br><span class='text-[10px] text-amber-400'>Info Pengalihan:</span><br><span class='text-[10px] leading-relaxed'>" + routingMsg + "</span>", "success");
                      } else {
                          showNotify("Berhasil Terkirim", res.message, "success");
                      }
                      
                      if (res.data) {
                          const logVirtual = {
                              id: res.data.id, tgl_pengajuan: res.data.tgl_pengajuan, nip: payload.nip,
                              jenis: payload.jenis, tgl_mulai: payload.tgl_mulai, tgl_selesai: payload.tgl_selesai,
                              alasan: payload.alasan, status: res.data.status, status_atasan: res.data.status,
                              atasan_nip: payload.atasan_nip, pejabat_nip: payload.pejabat_nip, sisa_cuti: res.data.durasi,
                              status_pejabat: res.data.status, alamat: payload.alamat, history: res.data.history, bukti_url: res.data.fileUrl,
                              plh_nip: payload.plh_nip
                          };
                          if (state && state.cutiLogs) state.cutiLogs.unshift(logVirtual);

                          if (state.usersList) {

                              if (targetUserIdx !== -1) {

        })
        .getSmartRouting(payload.atasan_nip, payload.pejabat_nip, payload.tgl_mulai, payload.tgl_selesai);

  function updateBadgePermohonan() {
      if (!state || !state.user) return;

      const isAdmin = (myRole === 'Admin' || myRole === 'Operator');

      let pendingCount = 0;
      let countAtasan = 0;
      let countPejabat = 0;

      // 1. Cuti
      if (state.cutiLogs) {
          state.cutiLogs.forEach(function(log) {
              const isPendingBatal = (log.status === 'Menunggu Pembatalan');
              if (log.status !== 'Menunggu' && !isPendingBatal) return;
              
              if (isAdmin) {
                  pendingCount++;
                  if (log.status_atasan === 'Menunggu') countAtasan++;
                  if (log.status_pejabat === 'Menunggu' && log.status_atasan === 'Disetujui') countPejabat++;
                  if (isPendingBatal) countPejabat++; 
              } else {
                  if (isPendingBatal && String(log.pejabat_nip) === myNip) pendingCount++;
                  else if (!isPendingBatal && String(log.atasan_nip) === myNip && log.status_atasan === 'Menunggu') pendingCount++;
                  else if (!isPendingBatal && String(log.pejabat_nip) === myNip && log.status_pejabat === 'Menunggu' && log.status_atasan === 'Disetujui') pendingCount++;
              }
          });
      }

      // 2. IK
      if (state.ikLogs) {
          state.ikLogs.forEach(function(log) {
              if (log.status !== 'Menunggu') return;
              if (isAdmin) { pendingCount++; countAtasan++; }
              else if (String(log.atasan_nip) === myNip) pendingCount++;
          });
      }

      // 3. IP
      if (state.ipLogs) {
          state.ipLogs.forEach(function(log) {
              if (log.status !== 'Menunggu') return;
              if (isAdmin) { pendingCount++; countAtasan++; }
              else if (String(log.atasan_nip) === myNip) pendingCount++;
          });
      }

      // 4. WFH
      if (state.wfhLogs) {
          state.wfhLogs.forEach(function(log) {
              if (log.status_rencana === 'Menunggu' || log.status_laporan === 'Menunggu') {

                  else if (String(log.atasan_nip) === myNip) pendingCount++;
              }
          });

  function calculateDuration(start, end) {
      const d1 = new Date(start); const d2 = new Date(end);
      let count = 0;
      for (let d = new Date(d1); d <= d2; d.setDate(d.getDate() + 1)) {
          if (d.getDay() !== 0 && d.getDay() !== 6) count++;
      }
      return count;
  }

  function updateLocalStateCuti(id, decision, notes) {
       const logIndex = state.cutiLogs.findIndex(l => String(l.id) === String(id));
       if (logIndex !== -1) {
           const log = state.cutiLogs[logIndex];
           const myNip = String(state.user.nip);
           const isAdmin = (state.user.role === 'Admin' || state.user.role === 'Operator');
           let isAtasan = (String(myNip) === String(log.atasan_nip));
           let isPejabat = (String(myNip) === String(log.pejabat_nip));
           let isBypassAtasan = (bypassMode === 'atasan' && isAdmin);
           let isBypassPejabat = (bypassMode === 'pejabat' && isAdmin);

           const statusBefore = log.status;

           if (log.status === 'Menunggu Pembatalan') {
               if (decision === 'Disetujui') log.status = 'DIBATALKAN';
               else log.status = 'Disetujui';
           } else {
               if ((isAtasan && isPejabat) || (isBypassAtasan && isBypassPejabat)) {
                   log.status_atasan = decision; log.status_pejabat = decision;
               } else {
                   if (isAtasan || isBypassAtasan) log.status_atasan = decision;
                   if (isPejabat || isBypassPejabat) log.status_pejabat = decision;
               }
               
               if (log.status_atasan === 'Ditolak' || log.status_pejabat === 'Ditolak') log.status = 'Ditolak';
               else if (log.status_atasan === 'Disetujui' && log.status_pejabat === 'Disetujui') log.status = 'Disetujui';
               else log.status = 'Menunggu';
           }

           let isRefundNeeded = false;
           if (log.status === 'Ditolak' && statusBefore !== 'Ditolak') isRefundNeeded = true;
           if (log.status === 'DIBATALKAN' && statusBefore !== 'DIBATALKAN') isRefundNeeded = true;

           if (isRefundNeeded) {
               const targetUserIdx = state.usersList.findIndex(u => String(u.nip) === String(log.nip));
               if (targetUserIdx !== -1) {
                   let tUser = state.usersList[targetUserIdx];
                   if (log.jenis === 'Tahunan') {
                       try {
                           let h = JSON.parse(log.history || "{}");
                           tUser.s_tahunan_2024 = Number(tUser.s_tahunan_2024 || 0) + Number(h.y24 || 0);
                           tUser.s_tahunan_2025 = Number(tUser.s_tahunan_2025 || 0) + Number(h.y25 || 0);
                           tUser.s_tahunan_2026 = Number(tUser.s_tahunan_2026 || 0) + Number(h.y26 || 0);
                       } catch(e) {}
                   } else if (log.jenis !== 'Dinas Luar') {
                       let field = "";
                       if(log.jenis === 'Sakit') field = 's_sakit';
                       else if(log.jenis === 'Alasan Penting' || log.jenis === 'Penting') field = 's_penting';
                       else if(log.jenis === 'Besar') field = 's_besar';
                       else if(log.jenis === 'Melahirkan') field = 's_melahirkan';
                       else if(log.jenis === 'CLTN') field = 's_cltn';
                       if (field !== "") tUser[field] = Number(tUser[field] || 0) + Number(log.sisa_cuti || 0);
                   }
                   if (String(log.nip) === String(state.user.nip)) state.user = Object.assign({}, tUser);
               }
           }
           fadeOutAndRefresh(id);
       }
  }

function addCutiLog(payload) {
  try {
    const ss = SpreadsheetApp.openById(PropertiesService.getScriptProperties().getProperty("TRANSKEP_DB_ID"));
    let sheetCuti = ss.getSheetByName('cuti_log');
    if(!sheetCuti) sheetCuti = ss.insertSheet('cuti_log');
    const sheetUser = ss.getSheetByName('user_data');
    
    let fileUrl = "";
    if (payload.fileData && payload.fileName && payload.fileMimeType) {
        try {
            const folderName = "Bukti Cuti TRANSKEP";
            let folders = DriveApp.getFoldersByName(folderName);
            let folder = folders.hasNext() ? folders.next() : DriveApp.createFolder(folderName);
            
            const cleanMime = payload.fileName.split('.').pop();
            const generateName = "Bukti_" + payload.jenis + "_" + payload.nip + "_" + new Date().getTime() + "." + cleanMime;
            
            const blob = Utilities.newBlob(Utilities.base64Decode(payload.fileData), payload.fileMimeType, generateName);
            const file = folder.createFile(blob);
            file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
            fileUrl = file.getUrl();
        } catch(e) { Logger.log("Gagal mengunggah file bukti: " + e.toString()); }
    }

    let d1 = new Date(payload.tgl_mulai); let d2 = new Date(payload.tgl_selesai);
    let durasi = 0;
    for (let d = new Date(d1); d <= d2; d.setDate(d.getDate() + 1)) {
        if (d.getDay() !== 0 && d.getDay() !== 6) durasi++;
    }

    const uData = sheetUser.getDataRange().getDisplayValues(); 
    let rowIndex = -1; let namaPegawaiPemohon = "";
    
    for (let i = 1; i < uData.length; i++) {
        if (String(uData[i][0]) === String(payload.nip)) { rowIndex = i + 1; namaPegawaiPemohon = uData[i][1]; break; }
    }
    if (rowIndex === -1) throw new Error("Pegawai tidak ditemukan.");

    let historyJson = { y24:0, y25:0, y26:0, dur: durasi };
    
    // Bypass potong saldo jika ini adalah usulan Dinas Luar
    if (payload.jenis !== 'Dinas Luar') {
        if (payload.jenis === 'Tahunan') {
            let s24 = Number(uData[rowIndex - 1][6]), s25 = Number(uData[rowIndex - 1][7]), s26 = Number(uData[rowIndex - 1][8]);
            let sisaPotong = durasi;
            
            if (s24 >= sisaPotong) { historyJson.y24 = sisaPotong; sheetUser.getRange(rowIndex, 7).setValue(s24 - sisaPotong); sisaPotong = 0; } 
            else { historyJson.y24 = s24; sisaPotong -= s24; sheetUser.getRange(rowIndex, 7).setValue(0); }
            
            if (sisaPotong > 0) {
                if (s25 >= sisaPotong) { historyJson.y25 = sisaPotong; sheetUser.getRange(rowIndex, 8).setValue(s25 - sisaPotong); sisaPotong = 0; } 
                else { historyJson.y25 = s25; sisaPotong -= s25; sheetUser.getRange(rowIndex, 8).setValue(0); }
            }
            if (sisaPotong > 0) {
                if (s26 >= sisaPotong) { historyJson.y26 = sisaPotong; sheetUser.getRange(rowIndex, 9).setValue(s26 - sisaPotong); } 
                else { throw new Error("Saldo cuti tahunan tidak mencukupi."); }
            }
        } else {
            const mapCol = { 'Sakit': 10, 'Alasan Penting': 11, 'Besar': 12, 'Melahirkan': 13, 'CLTN': 14 };
            const col = mapCol[payload.jenis];
            if (col) {
                let sCuti = Number(uData[rowIndex - 1][col - 1]);
                if (sCuti < durasi) throw new Error(`Saldo cuti ${payload.jenis} Anda tidak mencukupi.`);
                sheetUser.getRange(rowIndex, col).setValue(sCuti - durasi);
            }
        }
    }

    const uniquePrefix = payload.jenis === 'Dinas Luar' ? "DL-" : "CUTI-";
    const newId = uniquePrefix + new Date().getTime() + Math.floor(Math.random() * 1000);
    const tglForm = Utilities.formatDate(new Date(), 'Asia/Jakarta', 'dd/MM/yyyy');

function updateCutiStatus(id, decision, notes, userNip, bypassMode = null) {
  try {
    const ss = SpreadsheetApp.openById(PropertiesService.getScriptProperties().getProperty("TRANSKEP_DB_ID"));
    const sheetCuti = ss.getSheetByName('cuti_log');
    const sheetUser = ss.getSheetByName('user_data');
    
    const cutiData = sheetCuti.getDataRange().getValues(); 
    const userData = sheetUser.getDataRange().getDisplayValues(); 
    
    let rowIndex = -1; let logRecord = null;
    for (let i = 1; i < cutiData.length; i++) {
      if (String(cutiData[i][0]) === String(id)) { rowIndex = i + 1; logRecord = cutiData[i]; break; }
    }
    if (rowIndex === -1) throw new Error("Data transaksi cuti tidak ditemukan.");
    
    const nip = String(logRecord[2]); const jenis = String(logRecord[3]);
    const atasan_nip = String(logRecord[9]); const pejabat_nip = String(logRecord[10]);
    let cAtasan = String(logRecord[13]); let cPejabat = String(logRecord[14]); let cFinal = String(logRecord[7]);

    if (cFinal === 'Ditolak' || cFinal === 'DIBATALKAN') throw new Error("Transaksi ini sudah ditutup secara permanen.");
    
    let isAdmin = false; let isOperator = false; let namaPenyetuju = ""; 

    for (let i = 1; i < userData.length; i++) {
      if (String(userData[i][0]) === String(userNip)) { 
          if (String(userData[i][3]) === 'Admin') isAdmin = true;
          if (String(userData[i][3]) === 'Operator') isOperator = true;
          namaPenyetuju = userData[i][1];
          break;
      }
    }
    
    let isAtasan = (String(userNip) === atasan_nip); let isPejabat = (String(userNip) === pejabat_nip);
    let isBypassAtasan = (bypassMode === 'atasan' && (isAdmin || isOperator));
    let isBypassPejabat = (bypassMode === 'pejabat' && (isAdmin || isOperator));

    if (!isAtasan && !isPejabat && !isBypassAtasan && !isBypassPejabat) throw new Error("Akses Ditolak. Anda bukan atasan/pejabat terkait.");
    
    // ZETTBOT: Handler Khusus Usulan Pembatalan
    if (cFinal === 'Menunggu Pembatalan') {
       if (!isPejabat && !isBypassPejabat) throw new Error("Hanya Pejabat Berwenang yang dapat memproses Usulan Pembatalan Cuti.");
       
       if (decision === 'Disetujui') {
           const historyStr = String(logRecord[16] || "");
           let h24 = 0, h25 = 0, h26 = 0, hLain = 0;
           if (historyStr) { try { const h = JSON.parse(historyStr); h24 = h.y24 || 0; h25 = h.y25 || 0; h26 = h.y26 || 0; hLain = h.dur || 0; } catch(e) {} }
           
           let userRowIndex = -1;
           for (let i = 1; i < userData.length; i++) { if (String(userData[i][0]) === nip) { userRowIndex = i + 1; break; } }
           
           if (userRowIndex !== -1) {
              if (jenis === 'Tahunan') {
                 const c24 = Number(sheetUser.getRange(userRowIndex, 7).getValue() || 0); const c25 = Number(sheetUser.getRange(userRowIndex, 8).getValue() || 0); const c26 = Number(sheetUser.getRange(userRowIndex, 9).getValue() || 0);
                 sheetUser.getRange(userRowIndex, 7).setValue(c24 + h24); sheetUser.getRange(userRowIndex, 8).setValue(c25 + h25); sheetUser.getRange(userRowIndex, 9).setValue(c26 + h26);
              } else {
                 const mapKolom = { 'Sakit': 10, 'Alasan Penting': 11, 'Besar': 12, 'Melahirkan': 13, 'CLTN': 14 };
                 const col = mapKolom[jenis];
                 if (col) { const cVal = Number(sheetUser.getRange(userRowIndex, col).getValue() || 0); sheetUser.getRange(userRowIndex, col).setValue(cVal + hLain); }
              }
           }
           sheetCuti.getRange(rowIndex, 8).setValue('DIBATALKAN');
       } else {
           sheetCuti.getRange(rowIndex, 8).setValue('Disetujui');
       }
       
       let notesLama = String(logRecord[12] || "");
       let finalNotes = notesLama + "\n\n[JAWABAN PEMBATALAN]: " + notes;
       if (isBypassPejabat) finalNotes += " *(via Bypass oleh " + namaPenyetuju + ")*";
       sheetCuti.getRange(rowIndex, 13).setValue(finalNotes);
       
       SpreadsheetApp.flush();
       return { status: 'success', message: `Usulan Pembatalan telah di-${decision.toLowerCase()}.` };
    }

    // ZETTBOT FIX: Handler Cuti Normal & Dual Role
    let updated = false;
    let finalNotes = notes;
    
    // Jika user adalah Atasan DAN Pejabat sekaligus (Dual Role Ketua)
    if ((isAtasan && isPejabat) || (isBypassAtasan && isBypassPejabat)) {
        if (isBypassAtasan) finalNotes += "\n\n*(Disetujui via Bypass Dual-Role oleh " + namaPenyetuju + ")*";
        
        cAtasan = decision;
        cPejabat = decision;
        sheetCuti.getRange(rowIndex, 9).setValue(finalNotes); // catatan atasan
        sheetCuti.getRange(rowIndex, 14).setValue(decision);  // status atasan
        sheetCuti.getRange(rowIndex, 13).setValue(finalNotes); // catatan pejabat
        sheetCuti.getRange(rowIndex, 15).setValue(decision);  // status pejabat
        updated = true;
    } 
    // Logika Normal Tunggal
    else {
        if (isAtasan || isBypassAtasan) {
            if (cAtasan !== 'Menunggu') throw new Error("Berkas ini sudah diproses di tingkat Atasan.");
            cAtasan = decision; 
            if (isBypassAtasan) finalNotes += "\n\n*(Disetujui via Bypass Atasan oleh " + namaPenyetuju + ")*"; 
            
            sheetCuti.getRange(rowIndex, 9).setValue(finalNotes); 
            sheetCuti.getRange(rowIndex, 14).setValue(decision); 
            updated = true;
        }
        else if (isPejabat || isBypassPejabat) {
            if (cPejabat !== 'Menunggu') throw new Error("Berkas ini sudah diproses di tingkat Pejabat.");
            cPejabat = decision; 
            if (isBypassPejabat) finalNotes += "\n\n*(Disetujui via Bypass Pejabat oleh " + namaPenyetuju + ")*"; 
            
            sheetCuti.getRange(rowIndex, 13).setValue(finalNotes); 
            sheetCuti.getRange(rowIndex, 15).setValue(decision); 
            updated = true;
        }
    }
    
    if (!updated) throw new Error("Tidak ada aksi yang valid.");
    
    let finalStatus = 'Menunggu';
    if (cAtasan === 'Ditolak' || cPejabat === 'Ditolak') finalStatus = 'Ditolak';
    else if (cAtasan === 'Disetujui' && cPejabat === 'Disetujui') finalStatus = 'Disetujui';
    else if (cAtasan === 'Perubahan' || cPejabat === 'Perubahan') finalStatus = 'Perubahan';
    
    sheetCuti.getRange(rowIndex, 8).setValue(finalStatus);
    
    // Reverse Saldo Jika Penolakan Baru Terjadi
    if (finalStatus === 'Ditolak' || finalStatus === 'Perubahan') {
        const historyStr = String(logRecord[16] || "");
        let h24 = 0, h25 = 0, h26 = 0, hLain = 0;
        if (historyStr) { try { const h = JSON.parse(historyStr); h24 = h.y24 || 0; h25 = h.y25 || 0; h26 = h.y26 || 0; hLain = h.dur || 0; } catch(e) {} }
        
        let userRowIndex = -1;
        for (let i = 1; i < userData.length; i++) { if (String(userData[i][0]) === nip) { userRowIndex = i + 1; break; } }
        
        if (userRowIndex !== -1) {
           if (jenis === 'Tahunan') {
              const current24 = Number(sheetUser.getRange(userRowIndex, 7).getValue() || 0); const current25 = Number(sheetUser.getRange(userRowIndex, 8).getValue() || 0); const current26 = Number(sheetUser.getRange(userRowIndex, 9).getValue() || 0);
              sheetUser.getRange(userRowIndex, 7).setValue(current24 + h24); sheetUser.getRange(userRowIndex, 8).setValue(current25 + h25); sheetUser.getRange(userRowIndex, 9).setValue(current26 + h26);
           } else {
              const mapKolom = { 'Sakit': 10, 'Alasan Penting': 11, 'Besar': 12, 'Melahirkan': 13, 'CLTN': 14 };
              const col = mapKolom[jenis];
              if (col) { const currentVal = Number(sheetUser.getRange(userRowIndex, col).getValue() || 0); sheetUser.getRange(userRowIndex, col).setValue(currentVal + hLain); }
           }
        }
    }
    SpreadsheetApp.flush(); 
    return { status: 'success', message: `Berkas berhasil ${decision.toLowerCase()}.` };
  } catch (err) { return { status: 'error', message: err.toString() }; }
}

function processBulkApproval(bulkItems, decision, notes, userNip, bypassMode) {
    try {
        let successCount = 0;
        let failCount = 0;
        
        // Loop setiap ID dan teruskan ke fungsi persetujuan spesifiknya
        for (let i = 0; i < bulkItems.length; i++) {
            let item = bulkItems[i];
            try {
                if (item.type === 'Cuti') updateCutiStatus(item.id, decision, notes, userNip, bypassMode);
                else if (item.type === 'IK') updateIkStatus(item.id, decision, notes, userNip, bypassMode);
                else if (item.type === 'IP') updateIpStatus(item.id, decision, notes, userNip, bypassMode);
                else if (item.type === 'WFH') updateWfhStatus(item.id, item.stage, decision, notes, userNip, bypassMode);
                successCount++;
            } catch(err) {
                failCount++;
                Logger.log("Bulk Error pada ID " + item.id + ": " + err);
            }
        }
        
        return { 
            status: 'success', 
            message: `Proses persetujuan massal selesai. ${successCount} berkas berhasil diperbarui.` 
        };
    } catch(e) {
        return { status: 'error', message: e.toString() };
    }
}

function getSmartRouting(atasanNip, pejabatNip, tglMulai, tglSelesai) {
  try {
    const ss = SpreadsheetApp.openById(PropertiesService.getScriptProperties().getProperty("TRANSKEP_DB_ID"));
    const userSheet = ss.getSheetByName('user_data');
    const cutiSheet = ss.getSheetByName('cuti_log');
    const stSheet = ss.getSheetByName('st_log');

    const users = userSheet.getDataRange().getDisplayValues();
    const cutis = cutiSheet ? cutiSheet.getDataRange().getDisplayValues() : [];
    const sts = stSheet ? stSheet.getDataRange().getDisplayValues() : [];

    // 1. Map seluruh pegawai
    let userList = [];
    for(let i=1; i<users.length; i++) {
       userList.push({ nip: String(users[i][0]).trim(), nama: users[i][1], jabatan: users[i][2] });
    }

    // 2. Kumpulkan rekam jejak ketidakhadiran (Cuti & DL)
    let absences = [];
    
    // Cuti Disetujui
    for(let i=1; i<cutis.length; i++) {
        if(cutis[i][7] === 'Disetujui') {
             let hist = cutis[i][16] || '{}';
             let plh = "";
             try { let p = JSON.parse(hist); plh = p.plh_nip || ""; } catch(e) {}
             
             absences.push({
                 nip: String(cutis[i][2]).trim(),
                 start: new Date(cutis[i][4]),
                 end: new Date(cutis[i][5]),
                 plh_nip: plh,
                 jenis: "Cuti (" + cutis[i][3] + ")"
             });
        }
    }

    // ST / Dinas Luar Aktif
    for(let i=1; i<sts.length; i++) {
        if(sts[i][12] !== 'DIBATALKAN') {
             let pArr = [];
             try { pArr = JSON.parse(sts[i][10] || '[]'); } catch(e) {}
             pArr.forEach(p => {
                 absences.push({
                     nip: String(p.nip).trim(),
                     start: new Date(sts[i][7]),
                     end: new Date(sts[i][8]),
                     plh_nip: p.plh_nip || "",
                     jenis: "Dinas Luar"
                 });
             });
        }
    }

    // Helper: Cek Irisan Tanggal
    function checkOverlap(tMulai, tSelesai, aStart, aEnd) {
        let ts = new Date(tMulai); ts.setHours(0,0,0,0);
        let te = new Date(tSelesai); te.setHours(23,59,59,999);
        let as = new Date(aStart); as.setHours(0,0,0,0);
        let ae = new Date(aEnd); ae.setHours(23,59,59,999);
        return (ts <= ae) && (te >= as);
    }

    // Helper: Cari NIP berdasarkan Nama Jabatan
    function getRoleNip(roleName) {
        let found = userList.find(u => String(u.jabatan).toLowerCase() === roleName.toLowerCase());
        if(!found) found = userList.find(u => String(u.jabatan).toLowerCase().includes(roleName.toLowerCase()));
        return found ? found.nip : null;
    }

    let routingMessages = [];

    // Mesin Utama: Resolve (Eskalasi / Delegasi)
    function resolvePejabat(currentNip, typeName) {
        let resolvedNip = currentNip;
        let loopCount = 0; 
        
        while(loopCount < 3) { // Maks 3 lompatan agar tidak infinite loop
            loopCount++;
            let userObj = userList.find(u => u.nip === resolvedNip);
            if(!userObj) break;

            let isAbsent = false;
            let assignedPlh = "";
            let absenceReason = "";

            // Cek apakah Pejabat saat ini absen di tanggal tersebut
            for(let a of absences) {
                if(a.nip === resolvedNip && checkOverlap(tglMulai, tglSelesai, a.start, a.end)) {
                    isAbsent = true;
                    assignedPlh = a.plh_nip;
                    absenceReason = a.jenis;
                    break;
                }
            }

            if(!isAbsent) break; // Pejabat hadir, stop pencarian!

            let jab = String(userObj.jabatan).toLowerCase();
            
            // LOGIKA A & B: Eskalasi Naik Tingkat
            if(jab.includes("panitera muda") || jab.includes("panmud")) {
                let paniteraNip = getRoleNip("panitera");
                if(paniteraNip && paniteraNip !== resolvedNip) {
                    routingMessages.push(`- <b>${typeName}</b> dialihkan (naik) ke Panitera karena Panitera Muda sedang ${absenceReason}.`);
                    resolvedNip = paniteraNip;
                    continue; // Ulang loop untuk mengecek apakah Panitera juga absen
                }
            } 
            else if (jab.includes("kepala sub bagian") || jab.includes("kasubbag")) {
                let sekNip = getRoleNip("sekretaris");
                if(sekNip && sekNip !== resolvedNip) {
                    routingMessages.push(`- <b>${typeName}</b> dialihkan (naik) ke Sekretaris karena Kasubbag sedang ${absenceReason}.`);
                    resolvedNip = sekNip;
                    continue; 
                }
            }
            
            // LOGIKA C: Delegasi ke Plh 
            if(assignedPlh) {
                let plhObj = userList.find(u => u.nip === assignedPlh);
                let plhName = plhObj ? plhObj.nama : "Pejabat PLH";
                routingMessages.push(`- <b>${typeName}</b> dialihkan ke ${plhName} (selaku Plh), karena ${userObj.nama} sedang ${absenceReason}.`);
                resolvedNip = assignedPlh;
            } else {
                routingMessages.push(`- <b>Peringatan:</b> ${userObj.nama} sedang ${absenceReason} namun belum ada Plh di sistem.`);
                break; // Mentok, berhenti.
            }
        }
        
        return resolvedNip;
    }

    let finalAtasan = resolvePejabat(atasanNip, "Atasan Langsung");
    let finalPejabat = resolvePejabat(pejabatNip, "Pejabat Berwenang");

    return {
        status: 'success',
        data: {
            atasan_nip: finalAtasan,
            pejabat_nip: finalPejabat,
            pesan: routingMessages.length > 0 ? routingMessages.join("<br>") : ""
        }
    };

  } catch (err) {
    return { status: 'error', message: err.toString() };
  }
}
