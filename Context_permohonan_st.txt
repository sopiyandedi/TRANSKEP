  // Fungsi Helper Tanggal (Lokal - Backup if global is missed)
  function formatTglPlh(dateStr) {
      if (!dateStr) return "-";
      
      let d, m, y;
      
      if (dateStr.includes('/')) {
          // Format Database: DD/MM/YYYY (Contoh: 14/09/2026)
          const parts = dateStr.split('/');
          d = parseInt(parts[0], 10);
          m = parseInt(parts[1], 10) - 1; // Array bulan dimulai dari 0
          y = parseInt(parts[2], 10);
      } else if (dateStr.includes('-')) {
          // Format Input Form Date: YYYY-MM-DD (Contoh: 2026-09-14)
          const parts = dateStr.split('-');
          y = parseInt(parts[0], 10);
          m = parseInt(parts[1], 10) - 1;
          d = parseInt(parts[2], 10);
      } else {
          // Jika format kacau/tidak dikenali, kembalikan teks aslinya
          return dateStr;
      }

      // Validasi jika hasil belahan bukan angka
      if (isNaN(d) || isNaN(m) || isNaN(y)) return dateStr;

      const months = ["Januari", "Februari", "Maret", "April", "Mei", "Juni", "Juli", "Agustus", "September", "Oktober", "November", "Desember"];
      
      return `${d} ${months[m]} ${y}`;
  }

        if (start === end) return formatDateIndoCetak(start);

    // ZETTBOT FIX TAHAP 2: Perombakan fungsi format teks menggunakan Table Borderless
    // agar tulisan baris kedua rata menggantung persis seperti Microsoft Word.
    function formatTeksRapi(text) {
        if (!text) return "-";
        var lines = text.split('\n');
        var html = '<table style="width: 100%; border: none; font-size: 11pt; border-spacing: 0; padding: 0; margin: 0;">';
        
        for (var i = 0; i < lines.length; i++) {
            var line = lines[i].trim();
            if (line) {
                // Deteksi jika awal baris memiliki angka/huruf dan titik/kurung tutup (misal: 1. atau a.)
                var match = line.match(/^([0-9a-zA-Z]+[\.\)])\s+(.*)/);
                if (match) {
                    html += '<tr>' +
                               '<td style="width: 22px; vertical-align: top; padding: 0 0 2px 0; border: none;">' + match[1] + '</td>' +
                               '<td style="vertical-align: top; text-align: justify; padding: 0 0 2px 0; border: none;">' + match[2] + '</td>' +
                            '</tr>';
                } else {
                    html += '<tr>' +
                               '<td colspan="2" style="vertical-align: top; text-align: justify; padding: 0 0 2px 0; border: none;">' + line + '</td>' +
                            '</tr>';
                }
            }
        }
        html += '</table>';
        return html;
    }

  function attachSearchToInput(inputId, dropId, dataArr, onSelectCallback) {
      const inp = document.getElementById(inputId);
      const drop = document.getElementById(dropId);
      if(!inp || !drop) return;
      let focusIdx = 0;

      function renderList(filterTxt) {
          drop.innerHTML = '';
          let filtered = dataArr.filter(item => item.label.toLowerCase().includes(filterTxt.toLowerCase()));
          if(filtered.length === 0) {
              drop.innerHTML = `<div class="p-3 text-xs text-slate-500 italic">Tidak ditemukan...</div>`;
              drop.classList.remove('hidden');
              return;
          }
          filtered.forEach((item, idx) => {
              let div = document.createElement('div');
              let bgClass = (idx === focusIdx) ? 'bg-blue-600 text-white' : 'text-slate-300 hover:bg-slate-800';
              div.className = `px-3 py-2 text-[11px] cursor-pointer transition border-b border-slate-800/50 last:border-0 ${bgClass}`;
              div.innerHTML = item.html;
              div.onmousedown = function(e) {
                  e.preventDefault(); 
                  onSelectCallback(item);
                  drop.classList.add('hidden');
              };
              drop.appendChild(div);
          });
          drop.classList.remove('hidden');
      }

      inp.onfocus = () => { focusIdx = 0; renderList(inp.value); };
      inp.oninput = () => { focusIdx = 0; renderList(inp.value); };
      inp.onblur = () => { setTimeout(() => drop.classList.add('hidden'), 200); };

      inp.onkeydown = (e) => {
          let filtered = dataArr.filter(item => item.label.toLowerCase().includes(inp.value.toLowerCase()));
          let items = drop.children;

          if (e.key === 'ArrowDown') {
              e.preventDefault();
              focusIdx++;
              if (focusIdx >= filtered.length) focusIdx = filtered.length - 1;
              renderList(inp.value);
              if(items[focusIdx]) items[focusIdx].scrollIntoView({block: 'nearest'});
          } else if (e.key === 'ArrowUp') {
              e.preventDefault();
              focusIdx--;
              if (focusIdx < 0) focusIdx = 0;
              renderList(inp.value);
              if(items[focusIdx]) items[focusIdx].scrollIntoView({block: 'nearest'});
          } else if (e.key === 'Enter') {
              e.preventDefault(); 
              if (filtered.length > 0 && focusIdx < filtered.length) {
                  onSelectCallback(filtered[focusIdx]);
                  drop.classList.add('hidden');
              }
          }
      };
  }

        break;
      }

    if (rowIndex === -1) throw new Error("Data Surat Tugas tidak ditemukan untuk diperbarui.");

    // Ambil waktu pembuatan asli agar tidak berubah
    const originalWaktu = sheetST.getRange(rowIndex, 2).getValue();

    const updatedRow = [
      payload.id_st,

      payload.tujuan || '', 
      payload.tempat || '', 
      payload.tgl_mulai || '', 
      payload.tgl_selesai || '', 
      payload.jam || '',
      payload.peserta_json || '', 
      payload.pejabat_ttd || '',
      'DISETUJUI', // Standar status yang sama
      payload.data_ekstra_json || ''

  } catch (err) {
    return { status: 'error', message: err.toString() };
