function runDailyBackup() {
  try {
    // 1. Ambil ID Database Utama
    const dbId = PropertiesService.getScriptProperties().getProperty("TRANSKEP_DB_ID");
    if (!dbId) throw new Error("ID Database tidak ditemukan di sistem.");
    
    const dbFile = DriveApp.getFileById(dbId);
    const now = new Date();
    // Gunakan format hari-bulan-tahun
    const dateString = Utilities.formatDate(now, "Asia/Jakarta", "dd-MM-yyyy");
    
    // 2. Cari atau Buat Folder Backup di Drive Utama Anda
    const backupFolderName = "TRANSKEP BACKUPS";
    let backupFolder;
    const folders = DriveApp.getFoldersByName(backupFolderName);
    
    if (folders.hasNext()) {
      backupFolder = folders.next();
    } else {
      backupFolder = DriveApp.createFolder(backupFolderName);
    }
    
    // 3. Gandakan File (Backup Keseluruhan Database)
    const backupFileName = `[BACKUP] TRANSKEP_DB_${dateString}`;
    dbFile.makeCopy(backupFileName, backupFolder);
    Logger.log(`Backup harian sukses dibuat: ${backupFileName}`);
    
    // 4. Auto-Cleanup (Menghapus backup yang usianya sudah lebih dari 30 Hari)
    cleanupOldBackups(backupFolder);
    
  } catch (e) {
    Logger.log("ERROR Sistem Backup: " + e.toString());
  }
}

function cleanupOldBackups(folder) {
  const files = folder.getFiles();
  const now = new Date();
  const thirtyDaysInMs = 30 * 24 * 60 * 60 * 1000; // 30 Hari dalam hitungan Milidetik
  let deletedCount = 0;
  
  while (files.hasNext()) {
    const file = files.next();
    const dateCreated = file.getDateCreated();
    const ageInMs = now.getTime() - dateCreated.getTime();
    
    // Jika usia file lebih dari 30 hari
    if (ageInMs > thirtyDaysInMs) {
      file.setTrashed(true); // Memindahkan file ke tempat sampah (Trash)
      deletedCount++;
    }
  }
  
  if (deletedCount > 0) {
    Logger.log(`${deletedCount} file backup lama (lebih dari 30 hari) telah dibersihkan.`);
  }
}

function verifyLogin(nip, pass) {
  try {
    // ZETTBOT FIX: Direct Connection untuk mempercepat akses baca database
    let ss;
    try { ss = SpreadsheetApp.getActiveSpreadsheet(); } catch(e) {}
    if(!ss) ss = SpreadsheetApp.openById(PropertiesService.getScriptProperties().getProperty("TRANSKEP_DB_ID"));
    
    const sheet = ss.getSheetByName('user_data');
    const data = sheet.getDataRange().getDisplayValues();
    for (let i = 1; i < data.length; i++) {
      if (String(data[i][0]) === String(nip) && String(data[i][5]) === String(pass)) {
        const user = {
          nip: data[i][0], nama: data[i][1], jabatan: data[i][2], role: data[i][3],
          no_hp: data[i][4], s_tahunan_2024: data[i][6], s_tahunan_2025: data[i][7],
          s_tahunan_2026: data[i][8], s_sakit: data[i][9], s_penting: data[i][10],
          s_besar: data[i][11], s_melahirkan: data[i][12], s_cltn: data[i][13],
          pangkat: data[i][14], alamat: data[i][15]
        };

        // ZETTBOT SECURITY PATCH: GENERATE SESSION TOKEN
        const sessionToken = Utilities.getUuid(); // Buat UUID unik
        const cache = CacheService.getScriptCache();
        // Simpan token di server Google selama 8 Jam (28800 detik). Value-nya adalah NIP pemilik.
        cache.put(sessionToken, String(nip), 28800); 

        // Kembalikan token ke client
        return { status: 'success', data: user, token: sessionToken };
      }
    }
    return { status: 'error', message: 'NIP atau Sandi tidak valid' };
  } catch (e) { return { status: 'error', message: e.toString() }; }
}

       let parts = dateStr.includes('/') ? dateStr.split('/') : dateStr.split('-');

           let m = parts[1].padStart(2, '0');
           let y = parts[2];
           if (y.length === 2) y = "20" + y;

       }

       }
    }

    // ZETTBOT FIX: Quick count data PKP khusus untuk Badge Sidebar
    data.pendingPkpCount = 0;
    const sheetPkpDoc = ss.getSheetByName('pkp_dokumen');
    const sheetPckDoc = ss.getSheetByName('pck_dokumen');
    let pendingCount = 0;
    
    if (sheetPkpDoc) {
        const pkpData = sheetPkpDoc.getDataRange().getDisplayValues();
        const bawahanMap = {}; // Untuk memetakan atasan di PCK
        
        for(let i=1; i<pkpData.length; i++) {

            let dbAtasan = String(pkpData[i][7] || "").replace(/'/g, "").trim();

            if ((dbPenilai === String(nip) && status === "Menunggu Penilai") || 
                (dbAtasan === String(nip) && status === "Menunggu Atasan Penilai")) {
                pendingCount++;
            }
        }

        }
    }
    data.pendingPkpCount = pendingCount;

    return { status: 'success', data: data };
  } catch (e) { return { status: 'error', message: e.toString() }; }

function requestPasswordReset(no_wa) {
  try {
    let phone = String(no_wa).replace(/[^0-9]/g, '');
    if (phone.startsWith('0')) {
      phone = '62' + phone.substring(1);
    }

    const ss = SpreadsheetApp.openById(PropertiesService.getScriptProperties().getProperty("TRANSKEP_DB_ID"));
    const userSheet = ss.getSheetByName('user_data');
    if(!userSheet) throw new Error("Database user tidak ditemukan");
    
    const userData = userSheet.getDataRange().getDisplayValues(); 

    let userFound = null;
    for (let i = 1; i < userData.length; i++) {
      let userPhoneRaw = String(userData[i][4] || '').replace(/[^0-9]/g, '');
      if (userPhoneRaw.startsWith('0')) {
         userPhoneRaw = '62' + userPhoneRaw.substring(1);
      }
      
      if (userPhoneRaw === phone && phone.length > 8) {
        userFound = {
          nip: userData[i][0], nama: userData[i][1], no_wa: phone, pass: userData[i][5] 
        };
        break;
      }
    }

    if (!userFound) return { status: 'error', message: 'Nomor WhatsApp ini tidak terdaftar di dalam sistem pegawai. Silakan periksa kembali atau hubungi Admin.' };

    const logSheet = ss.getSheetByName('log_reset_pass');
    if(!logSheet) throw new Error("Database antrean belum siap. Minta Super Admin menjalankan initResetPasswordSheet() terlebih dahulu.");
    
    const uniqueId = 'REQ-' + new Date().getTime();
    const timestamp = Utilities.formatDate(new Date(), 'Asia/Jakarta', 'dd/MM/yyyy HH:mm:ss');

    logSheet.appendRow([uniqueId, timestamp, userFound.nip, userFound.nama, userFound.no_wa, userFound.pass]);
    SpreadsheetApp.flush();

    return { status: 'success', message: 'Permintaan berhasil dikirim. Admin akan segera memverifikasi dan mengirimkan kata sandi ke WA Anda.' };
  } catch (err) {
    return { status: 'error', message: err.toString() };
  }
}

}

function markResetAsDone(reqId) {
  try {
    const ss = SpreadsheetApp.openById(PropertiesService.getScriptProperties().getProperty("TRANSKEP_DB_ID"));
    const sheet = ss.getSheetByName('log_reset_pass');
    const data = sheet.getDataRange().getValues();

      if (String(data[i][0]) === String(reqId)) {
        rowIndex = i + 1;
        break;

      sheet.deleteRow(rowIndex);
      SpreadsheetApp.flush();
      return { status: 'success' };

    throw new Error("Antrean dengan ID tersebut tidak ditemukan atau sudah diproses orang lain.");
  } catch(e) {
    return { status: 'error', message: e.toString() };
  }
}

<script>
  console.log("🚀 ZETTBOT: Menginisialisasi sistem javascript...");

  // 1. GLOBAL STATE

    console.log("🚀 ZETTBOT: Halaman berhasil dimuat (DOMContentLoaded).");

    document.body.style.display = ''; 
    document.body.classList.remove('hidden');
    try {
      const activeNip = localStorage.getItem('transkep_active_nip');
      if (activeNip) {
        toggleAppVisibility(true);
        syncAllServerData();
      } else {
        toggleAppVisibility(false);

      toggleAppVisibility(false);
    }

  function toggleAppVisibility(isLoggedIn) {
    let loginSec = document.getElementById('login-section');
    let appCont = document.getElementById('app-container');
    const loader = document.getElementById('global-loader');
    
    if (loader) {
        loader.classList.add('hidden');
        loader.style.display = 'none';
    }

    if (!loginSec) {
        const passInput = document.querySelector('input[type="password"]');
        if (passInput) loginSec = passInput.closest('div.min-h-screen') || passInput.closest('form').parentElement.parentElement;
    }
    if (!appCont) {
         appCont = document.getElementById('main-content') || document.querySelector('main');
         if (appCont && appCont.tagName === 'MAIN') appCont = appCont.parentElement;
    }

    if (isLoggedIn) {
      if (loginSec) { loginSec.classList.add('hidden'); loginSec.style.display = 'none'; }
      if (appCont) { appCont.classList.remove('hidden'); appCont.style.display = ''; }
    } else {
      if (loginSec) { loginSec.classList.remove('hidden'); loginSec.style.display = ''; }
      if (appCont) { appCont.classList.add('hidden'); appCont.style.display = 'none'; }
    }
  }

  // 4. LOGIN & LOGOUT
  function handleLogin(event) {
    if (event) event.preventDefault();
    
    let nipInput = document.getElementById('loginNip'); 
    let passInput = document.getElementById('loginPass');
    if (!nipInput || !passInput) {
      nipInput = document.querySelector('input[type="text"], input[type="number"]');
      passInput = document.querySelector('input[type="password"]');
    }
    if (!nipInput || !passInput) return;

    const nip = nipInput.value.trim();
    const pass = passInput.value.trim();

    if (!nip || !pass) {
      showNotify("Perhatian", "NIP dan Kata Sandi wajib diisi!", "error");
      return;
    }

    showLoading("Memverifikasi Kredensial...");
    
    if (isGoogleEnv) {
      google.script.run
        .withSuccessHandler(function(response) {
          if (response && response.status === 'success') {
            localStorage.setItem('transkep_active_nip', response.data.nip);
            localStorage.setItem('transkep_session_token', response.token);
            
            toggleAppVisibility(true);
            syncAllServerData();
            showNotify("Login Berhasil", `Selamat datang kembali, ${response.data.nama}`, "success");
          } else {
            hideLoading();
            showNotify("Akses Ditolak", response ? response.message : "Tidak ada respon server", "error");
          }
        })
        .withFailureHandler(function(error) {
          hideLoading();
          showNotify("Error Server", error.message, "error");
        })
        .verifyLogin(nip, pass);
    }
  }

  function handleLogout() {

    localStorage.removeItem('transkep_last_menu');
    localStorage.removeItem('transkep_session_token');

  function updateSidebarBadges() {
      try {
          if (!state.user) return;

                  badgeSop.classList.add('hidden');
              }

              const involvedSks = state.skList.filter(sk => String(sk.nip).replace(/'/g, "").trim() === userNip);

                  const baseSkId = String(sk.id).replace(/'/g, "").trim().split('-').slice(0, 2).join('-');

          const badgeSk = document.getElementById('badge-sk-nav');

              if (unreadSkCount > 0) {
                  badgeSk.innerText = unreadSkCount > 99 ? '99+' : unreadSkCount;

              } else {
                  badgeSk.classList.add('hidden');

  function submitChangePassword(event) {
    event.preventDefault(); 
    
    const oldPass = document.getElementById('oldPassInput').value;
    const newPass = document.getElementById('newPassInput').value;
    const confirmPass = document.getElementById('confirmPassInput').value;

    if (newPass !== confirmPass) {
       showNotify("Validasi Gagal", "Konfirmasi sandi baru tidak sama. Silakan cek kembali.", "error");
       return;
    }
    
    if (newPass.length < 5) {
       showNotify("Sandi Terlalu Pendek", "Gunakan minimal 5 karakter untuk keamanan yang lebih baik.", "warning");
       return;
    }

    if (oldPass === newPass) {
       showNotify("Perhatian", "Sandi baru tidak boleh sama persis dengan sandi lama.", "warning");
       return;
    }

    showLoading("Mencocokkan kredensial di database...");

    const nip = state.user.nip; 

    google.script.run
      .withSuccessHandler(function(res) {
         hideLoading();
         if (res.status === 'success') {
            state.user.pass = newPass; 
            const listIndex = state.usersList.findIndex(u => String(u.nip) === String(nip));
            if (listIndex !== -1) state.usersList[listIndex].pass = newPass;

            closeChangePasswordModal();
            showNotify("Keamanan Diperbarui", "Sandi Anda telah sukses diubah. Ingat sandi baru Anda untuk sesi berikutnya.", "success");
         } else {
            showNotify("Ditolak Server", res.message, "error");
         }
      })
      .withFailureHandler(function(err) {
         hideLoading();
         showNotify("Error Jaringan", "Tidak dapat terhubung ke server. " + err.message, "error");
      })
      .changeMyPassword(nip, oldPass, newPass);
  }

function setupDatabase() {
  try {
    let ss = null;
    
    // 1. Coba deteksi apakah ini adalah bound-script (menyatu dengan sheet)
    try { 
      ss = SpreadsheetApp.getActiveSpreadsheet(); 
    } catch (boundError) {}
    
    // 2. Jika tidak terdeteksi (mungkin standalone), kita coba buat baru atau pakai yang ada
    if (!ss) {
      const props = PropertiesService.getScriptProperties();
      let ssId = props.getProperty("TRANSKEP_DB_ID");
      
      if (ssId) {
        try { ss = SpreadsheetApp.openById(ssId); } catch (openError) { ssId = null; }
      }
      
      if (!ss) {
        // Buat file baru jika benar-benar tidak ada
        ss = SpreadsheetApp.create("TRANSKEP_Database");
        props.setProperty("TRANSKEP_DB_ID", ss.getId());
      }
    }
    
    // 3. Simpan ID secara eksplisit ke Properties agar selalu bisa dipanggil oleh sistem
    if (ss) {
       PropertiesService.getScriptProperties().setProperty("TRANSKEP_DB_ID", ss.getId());
    } else {
       throw new Error("Gagal mendeteksi atau membuat spreadsheet database.");
    }
    
    // =====================================
    // SETUP TABEL USER (100% AMAN, TIDAK HAPUS DATA)
    // =====================================
    let userSheet = ss.getSheetByName('user_data');
