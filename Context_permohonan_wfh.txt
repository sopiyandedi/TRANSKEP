<script>
    function cetakLaporanWFH(id) {
        if (window.isGlobalPrinting) {
            showNotify("Proses Sibuk", "Sistem sedang menyiapkan dokumen. Harap tunggu.", "warning");
            return; 
        }
        
        if (!state || !state.user) {
            showNotify("Sistem Memuat", "Data belum siap.", "warning");
            return;
        }
        
        window.isGlobalPrinting = true;
        showLoading("Menyiapkan dokumen Laporan WFH...");
        
        if (typeof google !== 'undefined') {
            google.script.run
                .withSuccessHandler(function(res) {
                    if (res.status === 'success') {
                        renderTemplateCetakLaporanWFH(res.data);
                    } else {
                        hideLoading();
                        window.isGlobalPrinting = false;
                        showNotify("Gagal", res.message, "error");
                    }
                })
                .withFailureHandler(function(err) {
                    hideLoading();
                    window.isGlobalPrinting = false;
                    showNotify("Error Koneksi", err.toString(), "error");
                })
                .getRencanaWFHById(id);
        } else {
            setTimeout(() => {
                hideLoading();
                window.isGlobalPrinting = false;
                alert("[CANVAS MODE] Simulasi cetak Laporan WFH untuk ID: " + id);
            }, 1000);
        }
    }
    
    function renderTemplateCetakLaporanWFH(data) {
        try {
            const pemohon = state.usersList.find(u => String(u.nip) === String(data?.nip)) || { nama: '[Data Kosong]', jabatan: '-' };
            const atasan = state.usersList.find(u => String(u.nip) === String(data?.atasan_nip)) || { nama: '[Data Kosong]', jabatan: '-' };
            
            document.getElementById('cetakWfhLaporanId').innerText = data?.id || '-';
            document.getElementById('cetakWfhLaporanNama').innerText = pemohon.nama;
            document.getElementById('cetakWfhLaporanNip').innerText = data?.nip || '-';
            document.getElementById('cetakWfhLaporanJabatan').innerText = pemohon.jabatan;
            
            let tglMulai = data?.tanggal ? formatDateIndoRekap(data.tanggal) : '-';
            document.getElementById('cetakWfhLaporanTanggal').innerText = tglMulai;
            
            const tglBuat = data?.waktu ? data.waktu.split(' ')[0] : '-';
            document.getElementById('cetakWfhLaporanTanggalBuat').innerText = formatDateIndoRekap(tglBuat);
            
            document.getElementById('cetakWfhLaporanNamaPemohon').innerText = pemohon.nama;
            document.getElementById('cetakWfhLaporanNipPemohon').innerText = data?.nip || '-';
            
            document.getElementById('cetakWfhLaporanNamaAtasan').innerText = atasan.nama;
            document.getElementById('cetakWfhLaporanNipAtasan').innerText = data?.atasan_nip || '-';
            
            const tbody = document.getElementById('cetakWfhLaporanTabelBody');
            tbody.innerHTML = '';
            
            try {
                const hasilList = JSON.parse(data?.laporan_json || '[]');
                if (hasilList.length === 0) {
                    tbody.innerHTML = '<tr><td colspan="4" class="border border-black p-2 text-center italic text-slate-500">Laporan belum disusun atau data kosong.</td></tr>';
                } else {
                    hasilList.forEach((item, idx) => {
                        let statusHtml = item.status_capaian || '-';
                        if(statusHtml === "Selesai (Tercapai)") statusHtml = `<b>${statusHtml}</b>`;
                        
                        tbody.innerHTML += `
                            <tr>
                                <td class="border border-black p-2 text-center align-top">${idx + 1}</td>
                                <td class="border border-black p-2 align-top">
                                    <div class="font-bold">${item.deskripsi || '-'}</div>
                                    <div class="text-xs text-slate-600 mt-1">Qty Target: ${item.kuantitas || '-'}</div>
                                </td>
                                <td class="border border-black p-2 align-top">${item.realisasi || '-'}</td>
                                <td class="border border-black p-2 text-center align-top">${statusHtml}</td>
                            </tr>
                        `;
                    });
                }
            } catch (e) {
                tbody.innerHTML = '<tr><td colspan="4" class="border border-black p-2 text-center text-red-500 font-bold">Error membaca data laporan dari sistem.</td></tr>';
            }
            
            const printArea = document.getElementById('printAreaLaporanWFH');
            
            // ZETTBOT FIX: Pastikan form cetak lenyap dari layar aplikasi (disembunyikan via CSS inline)
            document.querySelectorAll('.print-only').forEach(el => {
                el.classList.add('hidden');
                el.style.display = 'none';
            });
            
            if(printArea) {
                printArea.classList.remove('hidden');
                printArea.style.display = 'none'; 
            }
            
            setTimeout(() => {
                hideLoading(); 
                window.focus(); 
                window.print();
                
                setTimeout(() => {
                    if(printArea) {
                        printArea.classList.add('hidden');
                        printArea.style.display = '';
                    }
                    window.isGlobalPrinting = false;
                }, 1500); 

            }, 800);
            
        } catch(e) {
            hideLoading();
            window.isGlobalPrinting = false;
            showNotify("Error Render", "Terjadi kesalahan saat memuat template cetak Laporan WFH.", "error");
            console.error(e);
        }
    }
</script>

    function openModalWFH(editId = null) {

            return;
        }
        
        currentEditIdWFH = editId;

        // ZETTBOT FIX: Evaluasi Routing Role-Based & Alert Plh
        const selAtasan = document.getElementById('wfhAtasan');
        const alertPlh = document.getElementById('alertPlhWFH');
        selAtasan.innerHTML = '<option value="">-- Pilih Penanggung Jawab --</option>';
        
        let atasanOpts = [];
        let adaPlh = false;
        const j = state.user.jabatan.toLowerCase();

        function getPegawai(keyword) {
            return state.usersList.filter(x => String(x.nip) !== String(state.user.nip) && x.jabatan.toLowerCase().includes(keyword));
        }

        // LOGIKA A: Hakim, Wakil Ketua, Ketua -> ke Ketua
        if (j.includes("hakim") || j.includes("wakil ketua") || j === "ketua") {
            let ketua = getPegawai("ketua").find(x => x.jabatan.toLowerCase() === "ketua");
            if (ketua && typeof isPegawaiAbsent === 'function' && !isPegawaiAbsent(ketua.nip)) {
                atasanOpts.push(ketua);
            } else {
                adaPlh = true;
                let wakil = getPegawai("wakil ketua").find(x => x.jabatan.toLowerCase() === "wakil ketua");
                if (wakil && typeof isPegawaiAbsent === 'function' && !isPegawaiAbsent(wakil.nip)) atasanOpts.push(wakil);
                else { 
                    let hakims = getPegawai("hakim").filter(x => typeof isPegawaiAbsent === 'function' ? !isPegawaiAbsent(x.nip) : true); 
                    atasanOpts.push(...hakims); 
                }
            }
        }
        // LOGIKA B: Kepaniteraan (Panmud/PP/Jurusita) -> ke Panitera
        else if (j.includes("panitera muda") || j.includes("panitera pengganti") || j.includes("jurusita") || j.includes("juru sita")) {

            }
        }

            else {

        else if (alertPlh) alertPlh.classList.add('hidden');

    function submitFormWFH(e) {
        e.preventDefault();
        
        if(!state.user) {
            showNotify("Sesi Berakhir", "Harap login ulang.", "error");
            return;
        }

        const tglPilih = document.getElementById('wfhTanggal').value;
        
        if (new Date(tglPilih).getDay() !== 5) {
            showNotify("Tanggal Ditolak", "Sesuai aturan, pelaksanaan WFH hanya diizinkan pada hari Jumat.", "warning");
            return;
        }

        let targetList = [];
        const rows = document.querySelectorAll('.target-row');
        rows.forEach(row => {
            const desk = row.querySelector('.target-deskripsi').value;
            const qty = row.querySelector('.target-kuantitas').value;
            if(desk.trim() !== '' && qty.trim() !== '') {
                targetList.push({ deskripsi: desk, kuantitas: qty, realisasi: '', status_capaian: '' });
            }
        });

        if(targetList.length === 0) {
            showNotify("Validasi Gagal", "Target kerja tidak boleh kosong.", "error");
            return;
        }
        
        const payload = {
            id: currentEditIdWFH,
            nip: state.user.nip,
            tanggal: tglPilih,
            atasan_nip: document.getElementById('wfhAtasan').value,
            rencana_json: JSON.stringify(targetList),
        };
        
        showLoading(currentEditIdWFH ? "Memperbarui Rencana WFH..." : "Mengirimkan Rencana WFH...");
        
        if (typeof google !== 'undefined') {
            // ZETTBOT FIX: Simpan handler ke variabel runner
            const runner = google.script.run
              .withSuccessHandler(function(res) {
                 hideLoading();
                 if(res.status === 'success') {
                    closeModalWFH();
                    showNotify("Berhasil Disimpan", res.message, "success");
                    
                    if (currentEditIdWFH) {
                        const idx = state.wfhLogs.findIndex(l => String(l.id) === String(currentEditIdWFH));
                        if (idx !== -1) {
                            state.wfhLogs[idx].tanggal = payload.tanggal;
                            state.wfhLogs[idx].atasan_nip = payload.atasan_nip;
                            state.wfhLogs[idx].rencana_json = payload.rencana_json;
                            state.wfhLogs[idx].status_rencana = 'Menunggu';
                        }
                    } else {
                        const virtualLog = {
                            id: (res.data && res.data.id) ? res.data.id : "WFH-VIRTUAL-" + Date.now(),
                            tgl_pengajuan: new Date().toLocaleDateString('id-ID'), 
                            nip: payload.nip,
                            tanggal: payload.tanggal,
                            rencana_json: payload.rencana_json,
                            laporan_json: "",
                            atasan_nip: payload.atasan_nip,
                            status_rencana: 'Menunggu',
                            status_laporan: 'Belum Lapor',
                            z_type: 'WFH',
                            z_sort: Date.now() 
                        };
                        
                        if (state && state.wfhLogs) {
                            state.wfhLogs.unshift(virtualLog);
                        }
                    }

                    if (typeof renderPermohonanView === 'function') {
                        renderPermohonanView();
                    }
                 } else {
                    showNotify("Gagal", res.message, "error");
                 }
              })
              .withFailureHandler(function(err) {
                 hideLoading();
                 showNotify("Error Jaringan", err.toString(), "error");
              });
              
            // ZETTBOT FIX: Eksekusi runner sesuai kondisi
            if (currentEditIdWFH) {
                runner.updateRencanaWfhLog(payload);
            } else {
                runner.submitRencanaWFH(payload);
            }
        } else {
            setTimeout(() => {
                hideLoading();
                closeModalWFH();
                alert("[CANVAS MODE] Simulasi berhasil!");
            }, 1000);
        }
    }

  try {
    const ss = SpreadsheetApp.openById(PropertiesService.getScriptProperties().getProperty("TRANSKEP_DB_ID"));
    let sheet = ss.getSheetByName('wfh_log');
    
    if(!sheet) {
        sheet = ss.insertSheet('wfh_log');

        sheet.getRange("A1:J1").setFontWeight("bold").setBackground("#b45309").setFontColor("#ffffff");
        sheet.setFrozenRows(1);
    }
    
    const timestamp = Utilities.formatDate(new Date(), 'Asia/Jakarta', 'dd/MM/yyyy HH:mm:ss');
    const newId = "WFH-" + new Date().getTime() + Math.floor(Math.random() * 1000);
    
    sheet.appendRow([
        newId,
        timestamp,
        payload.nip,
        payload.tanggal,
        payload.rencana_json,
        "", // Laporan JSON kosong di awal
        payload.atasan_nip,
        'Menunggu',
        'Belum Lapor',
        ''
    ]);
    
    SpreadsheetApp.flush();
    return { status: 'success', message: 'Rencana WFH berhasil diajukan.', data: { id: newId } };
  } catch (err) { 
    return { status: 'error', message: err.toString() }; 
  }
}

function updateWfhStatus(id, stage, decision, notes, userNip, bypassMode = null) {
  try {
    const ss = SpreadsheetApp.openById(PropertiesService.getScriptProperties().getProperty("TRANSKEP_DB_ID"));
    const sheet = ss.getSheetByName('wfh_log');
    const data = sheet.getDataRange().getValues();
    
    let rowIndex = -1;
    for (let i = 1; i < data.length; i++) {
      if (String(data[i][0]) === String(id)) { rowIndex = i + 1; break; }
    }
    if (rowIndex === -1) throw new Error("Data WFH tidak ditemukan.");
    
    let finalNotes = notes;
    if (bypassMode) finalNotes += "\n*(Disetujui via Bypass Atasan)*";
    
    let oldNotes = String(data[rowIndex-1][9] || "");
    let combinedNotes = oldNotes + "\n[" + stage + "]: " + finalNotes;
    
    if (stage === 'Rencana') {
        sheet.getRange(rowIndex, 8).setValue(decision); 
    } else if (stage === 'Laporan') {
        sheet.getRange(rowIndex, 9).setValue(decision); 
    }
    
    sheet.getRange(rowIndex, 10).setValue(combinedNotes.trim()); 
    
    SpreadsheetApp.flush();
    return { status: 'success', message: `Usulan ${stage} WFH berhasil diproses.` };
  } catch(e) { 
    return { status: 'error', message: e.toString() }; 
  }
}

function deleteWfhLog(id, forceDelete = false) {
  try {
    const ss = SpreadsheetApp.openById(PropertiesService.getScriptProperties().getProperty("TRANSKEP_DB_ID"));
    const sheet = ss.getSheetByName('wfh_log');
    const data = sheet.getDataRange().getValues();
    
    let rowIndex = -1;
    for (let i = 1; i < data.length; i++) {
      if (String(data[i][0]) === String(id)) { 
        if (!forceDelete && data[i][7] !== 'Menunggu' && data[i][8] !== 'Menunggu') {
            throw new Error("Akses Ditolak: Hanya Rencana/Laporan berstatus Menunggu yang dapat ditarik.");
        }
        rowIndex = i + 1; 
        break; 
      }
    }
    if (rowIndex === -1) throw new Error("Data tidak ditemukan di sistem.");
    
    sheet.deleteRow(rowIndex);
    SpreadsheetApp.flush();
    return { status: 'success', message: 'Pengajuan WFH berhasil ditarik dan dihapus.' };
  } catch (err) { return { status: 'error', message: err.toString() }; }
}
