function updateTmtStatus(nip, type, newStatus) {
  try {
    const ss = SpreadsheetApp.openById(PropertiesService.getScriptProperties().getProperty("TRANSKEP_DB_ID"));
    const sheet = ss.getSheetByName('user_data');
    const data = sheet.getDataRange().getValues();
    const headers = data[0];

    let rowIndex = -1;
    for(let i=1; i<data.length; i++) {
      if(String(data[i][0]) === String(nip)) { rowIndex = i+1; break; }
    }
    if(rowIndex === -1) throw new Error("Pegawai tidak ditemukan di database.");

    let colStatus, colTgl;
    if(type === 'KGB') {
      colStatus = headers.indexOf('Status_KGB') + 1;
      colTgl = headers.indexOf('Tgl_Selesai_KGB') + 1;
    } else if(type === 'Pangkat' || type === 'KNP') {
      colStatus = headers.indexOf('Status_KNP') + 1;
      colTgl = headers.indexOf('Tgl_Selesai_KNP') + 1;
    } else if(type === 'Pensiun') { // ZETTBOT FIX: Dukungan Status Pensiun
      colStatus = headers.indexOf('Status_Pensiun') + 1;
      colTgl = headers.indexOf('Tgl_Selesai_Pensiun') + 1;
    } else {
      throw new Error("Tipe peringatan tidak valid.");
    }

    if(colStatus === 0) throw new Error("Kolom belum diinisialisasi. Pastikan database lengkap.");

    let todayStr = Utilities.formatDate(new Date(), 'Asia/Jakarta', 'yyyy-MM-dd');
    let valTgl = (newStatus === 'Selesai') ? todayStr : '';
    let valStatus = (newStatus === 'Menunggu') ? '' : newStatus; 

    sheet.getRange(rowIndex, colStatus).setValue(valStatus);
    sheet.getRange(rowIndex, colTgl).setValue(valTgl);

    SpreadsheetApp.flush();
    return { status: 'success', message: 'Status berhasil diperbarui' };
  } catch (e) {
    return { status: 'error', message: e.toString() };
  }
}

function forwardTmtDateByFrontend(nip, type, newDateStr) {
  try {
    const ss = SpreadsheetApp.openById(PropertiesService.getScriptProperties().getProperty("TRANSKEP_DB_ID"));
    const sheet = ss.getSheetByName('user_data');
    const data = sheet.getDataRange().getValues();
    const headers = data[0];

    let rowIndex = -1;
    for(let i=1; i<data.length; i++) {
      if(String(data[i][0]) === String(nip)) { rowIndex = i+1; break; }
    }
    if(rowIndex === -1) return { status: 'error', message: 'Pegawai tidak ditemukan' };

    let colTmt, colStatus, colTgl;
    if(type === 'KGB') {
      colTmt = headers.indexOf('TMT_KGB') + 1;
      colStatus = headers.indexOf('Status_KGB') + 1;
      colTgl = headers.indexOf('Tgl_Selesai_KGB') + 1;
    } else if(type === 'Pangkat' || type === 'KNP') {
      colTmt = headers.indexOf('TMT_KNP') + 1;
      colStatus = headers.indexOf('Status_KNP') + 1;
      colTgl = headers.indexOf('Tgl_Selesai_KNP') + 1;
    } else {
      return { status: 'error' };
    }

    sheet.getRange(rowIndex, colTmt).setValue(newDateStr);
    sheet.getRange(rowIndex, colTmt).setNumberFormat('@'); 
    
    if (colStatus > 0) {
        sheet.getRange(rowIndex, colStatus).setValue('');
        sheet.getRange(rowIndex, colTgl).setValue('');
    }

    SpreadsheetApp.flush();
    return { status: 'success' };
  } catch(e) {
    return { status: 'error', message: e.toString() };
  }
}

  function formatYMD(dateObj) {
      let d = new Date(dateObj), month = '' + (d.getMonth() + 1), day = '' + d.getDate(), year = d.getFullYear();

      if (day.length < 2) day = '0' + day;

  function parseTanggalLokal(tglStr) {
      if (!tglStr) return null;
      tglStr = String(tglStr).trim();
      
      if (!/[a-zA-Z]/.test(tglStr)) {
          let parts = tglStr.split(/[\/\-]/);
          if (parts.length === 3) {
              let p0 = parseInt(parts[0], 10);
              let p1 = parseInt(parts[1], 10);
              let p2 = parseInt(parts[2], 10);
              
              if (!isNaN(p0) && !isNaN(p1) && !isNaN(p2)) {
                  if (p2 > 1000) { 
                      return new Date(p2, p1 - 1, p0); 
                  } else if (p0 > 1000) { 
                      return new Date(p0, p1 - 1, p2); 
                  }
              }
          }
      }

      let cleanStr = tglStr.replace(/-/g, ' ');
      let d = new Date(cleanStr);
      if (!isNaN(d.getTime())) return d;
      
      return null; 
  }

  function createWarningObj(user, type, targetDate, today, dbStatus) {
      let diffTime = targetDate.getTime() - today.getTime();
      let diffDays = Math.ceil(diffTime / (1000 * 3600 * 24));
      
      let status = 'Menunggu', statusClass = 'text-amber-400 border-amber-500/30 bg-amber-500/10', targetText = '', isCompletedOrProcessing = false, targetClass = 'text-indigo-400'; 
      let checkStatus = (dbStatus || '').toUpperCase();

      if (checkStatus === 'PROSES') {
          status = 'Sedang Diproses'; statusClass = 'text-blue-400 border-blue-500/30 bg-blue-500/10 shadow-[0_0_10px_rgba(59,130,246,0.2)] font-bold'; isCompletedOrProcessing = true; targetText = 'Berkas Terkirim'; targetClass = 'text-blue-400';
      } else if (checkStatus === 'SELESAI') {
          status = 'Selesai'; statusClass = 'text-emerald-400 border-emerald-500/30 bg-emerald-500/10 shadow-[0_0_10px_rgba(16,185,129,0.2)] font-bold'; isCompletedOrProcessing = true; targetText = 'Berkas Diterima'; targetClass = 'text-emerald-400';
      } else {
          if (diffDays < 0) { status = 'Terlewat'; statusClass = 'text-rose-400 border-rose-500/50 bg-rose-500/10 shadow-[0_0_10px_rgba(244,63,94,0.2)] font-bold'; targetClass = 'text-rose-400'; } 
          else if (diffDays <= 90) { status = 'Segera'; statusClass = 'text-orange-400 border-orange-500/50 bg-orange-500/10 font-bold'; }
      }
      
      if (!isCompletedOrProcessing) {
          if (diffDays < 0) targetText = 'OVERDUE';
          else {
              let diffY = Math.floor(diffDays / 365), diffM = Math.floor((diffDays % 365) / 30);
              if (diffY > 0) targetText += diffY + ' Thn ';
              if (diffM > 0) targetText += diffM + ' Bln';
              if (!targetText) targetText = diffDays + ' Hari';
          }
      }
      return { user: user, type: type, rawDate: targetDate, dateStr: formatTanggalIndoDashboard(targetDate), status: status, statusClass: statusClass, targetText: targetText, targetClass: targetClass, month: targetDate.getMonth(), year: targetDate.getFullYear(), dbStatus: dbStatus || '' };
