  function parseDateToHTML5(sheetDateStr) {
    if (!sheetDateStr) return "";
    // Jika format sudah YYYY-MM-DD, kembalikan langsung
    if (sheetDateStr.includes('-') && sheetDateStr.split('-')[0].length === 4) return sheetDateStr;
    
    // Konversi dari DD/MM/YY atau DD/MM/YYYY
    const parts = sheetDateStr.includes('/') ? sheetDateStr.split('/') : sheetDateStr.split('-');
    if (parts.length === 3) {
        let day = parts[0].padStart(2, '0');
        let month = parts[1].padStart(2, '0');
        let year = parts[2];
        // Jika tahun cuma 2 digit (misal: 28), asumsikan tahun 2000-an (2028)
        if (year.length === 2) year = "20" + year;
        return `${year}-${month}-${day}`;
    }
    return "";
  }

  function handleImportPegawai(event) {
    const file = event.target.files[0];
    if (!file) return;
    
    const reader = new FileReader();
    reader.onload = function(e) {
        try {
            const data = new Uint8Array(e.target.result);
            const workbook = XLSX.read(data, {type: 'array'});
            
            // Ambil sheet pertama saja
            const firstSheetName = workbook.SheetNames[0];
            const worksheet = workbook.Sheets[firstSheetName];
            
            // Konversi ke JSON
            const json = XLSX.utils.sheet_to_json(worksheet, { defval: "" }); // defval mencegah undefined
            
            if (json.length === 0) {
                showNotify('Import Gagal', 'File Excel yang diunggah kosong.', 'error');
                return;
            }

            // Validasi keberadaan Header Krusial (Mencegah salah upload file)
            if (!json[0].hasOwnProperty('NIP') || !json[0].hasOwnProperty('Nama')) {
                showNotify('Import Ditolak', 'Format Excel tidak sesuai. Wajib menggunakan format hasil Backup (Memiliki header NIP & Nama).', 'error');
                return;
            }

            // Tampilkan Konfirmasi via Custom Modal Zettbot
            showCustomConfirm(
                "Konfirmasi Sinkronisasi", 
                `Ditemukan ${json.length} data profil di dalam file. Sistem akan menambahkan yang baru dan memperbarui (menimpa) NIP yang sudah ada. Lanjutkan?`, 
                function() {
                    // Eksekusi Import
                    showLoading("Memasukkan data ke database...");
                    google.script.run
                        .withSuccessHandler(res => {
                            hideLoading();
                            document.getElementById('importPegawaiInput').value = ''; // Reset File Input
                            if(res.status === 'success') {
                                showNotify('Import Selesai', res.message, 'success');
                                syncAllServerData(); // Reload UI otomatis
                            } else {
                                showNotify('Import Gagal', res.message, 'error');
                            }
                        })
                        .withFailureHandler(err => {
                            hideLoading();
                            document.getElementById('importPegawaiInput').value = ''; // Reset File Input
                            showNotify('Error Sistem', err.toString(), 'error');
                        })
                        .importBulkUsers(json);
                }
            );

        } catch(error) {
            document.getElementById('importPegawaiInput').value = ''; // Reset
            showNotify('Import Gagal', 'Dokumen rusak atau tidak terbaca sebagai Excel murni.', 'error');
        }
    };
    reader.readAsArrayBuffer(file);
  }

  function filterPegawaiTable() {
    const input = document.getElementById('pegSearchInput');
    const filter = input ? input.value.toLowerCase().trim() : "";
    const tbody = document.getElementById('tblManajemenBody');
    if (!tbody) return;
    
    const rows = tbody.getElementsByTagName('tr');
    if (rows.length === 1 && rows[0].cells.length === 1) return;

    for (let i = 0; i < rows.length; i += 2) {
      const mainRow = rows[i];
      const detailRow = rows[i + 1];
      if (!mainRow) continue;

      const nameCell = mainRow.cells[1];
      const jabatanCell = mainRow.cells[2];

      if (nameCell || jabatanCell) {
        const nameText = nameCell ? nameCell.textContent || nameCell.innerText : "";
        const jabatanText = jabatanCell ? jabatanCell.textContent || jabatanCell.innerText : "";
        
        if (nameText.toLowerCase().indexOf(filter) > -1 || jabatanText.toLowerCase().indexOf(filter) > -1) {
          mainRow.style.display = "";
        } else {
          mainRow.style.display = "none";
          if (detailRow) {
            detailRow.style.display = "none";
            const nip = mainRow.id ? mainRow.id.replace('peg-main-row-', '') : '';
            if (nip) {
              const chevron = document.getElementById(`chevron-peg-${nip}`);
              if (chevron) chevron.classList.remove('rotate-180');
              detailRow.classList.add('hidden');
            }
          }
        }
      }
    }
  }

function addUser(payload) {
  try {
    const ss = SpreadsheetApp.openById(PropertiesService.getScriptProperties().getProperty("TRANSKEP_DB_ID"));
    const sheet = ss.getSheetByName('user_data');
    const data = sheet.getDataRange().getValues();
    
    // Cek apakah NIP sudah ada
    for (let i = 1; i < data.length; i++) {
      if (String(data[i][0]) === String(payload.nip)) {
        throw new Error("NIP sudah terdaftar di database.");
      }
    }
    
    const newRow = [
      payload.nip, payload.nama, payload.jabatan, payload.role, payload.no_hp, payload.pass,
      payload.s_tahunan_2024, payload.s_tahunan_2025, payload.s_tahunan_2026,
      payload.s_sakit, payload.s_penting, payload.s_besar, payload.s_melahirkan, payload.s_cltn,
      payload.pangkat, payload.alamat, payload.tmt_kgb, payload.tmt_knp, payload.bup
    ];
    
    sheet.appendRow(newRow);
    const lastRow = sheet.getLastRow();
    sheet.getRange(lastRow, 1).setNumberFormat('@'); // Set NIP jadi plain text agar tidak rusak (0 di depan hilang)
    
    SpreadsheetApp.flush();
    return { status: 'success', message: 'Data pegawai baru berhasil ditambahkan.' };
  } catch (err) {
    return { status: 'error', message: err.toString() };
  }
}

function updateUser(payload) {
  try {
    const ss = SpreadsheetApp.openById(PropertiesService.getScriptProperties().getProperty("TRANSKEP_DB_ID"));
    const sheet = ss.getSheetByName('user_data');
