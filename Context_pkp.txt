function initPkpModule() {
    try {
        console.log("🚀 ZETTBOT: Menjalankan initPkpModule()...");
        
        var role = state.user ? state.user.role : '';
        var cleanUserNip = state.user ? String(state.user.nip).replace(/'/g, "").trim() : "";

        // ZETTBOT FIX: Hak Akses Ketat Tab Lapis 1 (Riwayat, Verifikasi, Rekap)
        var btnVerifikasi = document.getElementById("btn-pkp-verifikasi");
        var btnRekap = document.getElementById("btn-pkp-rekap");
        var btnRiwayat = document.getElementById("btn-pkp-riwayat");
        var tabLapis1Container = btnRiwayat ? btnRiwayat.parentElement : null;
        
        // 1. Reset (Sembunyikan semua secara default agar container hilang untuk Pegawai)
        if(btnVerifikasi) btnVerifikasi.classList.add("hidden");
        if(btnRekap) btnRekap.classList.add("hidden");
        if(tabLapis1Container) tabLapis1Container.classList.add("hidden");

        // 2. Munculkan kembali HANYA JIKA rolenya memenuhi syarat
        if (role === 'Admin' || role === 'Operator' || role === 'Atasan Langsung' || role === 'Pejabat Berwenang') {
            if(tabLapis1Container) tabLapis1Container.classList.remove("hidden");
            if(btnVerifikasi) btnVerifikasi.classList.remove("hidden");
        }
        if (role === 'Admin' || role === 'Operator') {
            if(btnRekap) btnRekap.classList.remove("hidden");
        }
        
        var filterTahun = document.getElementById("filterTahunPCK");
        if (filterTahun && filterTahun.options.length === 0) {
            var currentYear = new Date().getFullYear();
            filterTahun.innerHTML = "<option value='" + (currentYear + 1) + "'>Tahun " + (currentYear + 1) + "</option>" +
                                    "<option value='" + currentYear + "' selected>Tahun " + currentYear + "</option>" +
                                    "<option value='" + (currentYear - 1) + "'>Tahun " + (currentYear - 1) + "</option>";
        }

        showLoading("Sinkronisasi Modul PKP...");
        if (typeof google !== 'undefined' && google.script) {
            google.script.run
                .withSuccessHandler(function(resPK) {
                    state.pkpDokumen = resPK || [];
                    
                    google.script.run
                        .withSuccessHandler(function(resPCK) {
                            hideLoading();
                            state.pckDokumen = resPCK || [];
                            
                            switchMainTabPKP('PK'); 
                            if (typeof updateBadgeVerifikasiPKP === 'function') updateBadgeVerifikasiPKP();
                        })
                        .withFailureHandler(function(e) {
                            hideLoading();
                            showNotify("Gagal Tarik PCK", e.message, "error");
                        })
                        .getDataDokumenPCK(cleanUserNip, role);
                })
                .withFailureHandler(function(e) {
                    hideLoading();
                    showNotify("Gagal Tarik PKP", e.message, "error");
                })
                .getDataDokumenPKP(cleanUserNip, role);
        } else {
            setTimeout(function() {
                hideLoading();
                switchMainTabPKP('PK'); 
            }, 500);
        }
    } catch (e) {
        console.error("Error di initPkpModule:", e);
    }
}

function cetakDokumen(jenis, id) {
    if (jenis === 'PK') {
        showNotify("Persiapan Cetak PK", "Mengkompilasi data Target Tahunan...", "success");
        if (typeof google !== 'undefined' && google.script) {
            google.script.run
                .withSuccessHandler(function(res) {
                    if (res && res.success) renderPrintTemplatePK(res);
                    else showNotify("Gagal Cetak", "Detail dokumen tidak ditemukan.", "error");
                })
                .getDetailPKP(id);
        }
    } else if (jenis === 'PCK') {
        showNotify("Persiapan Cetak PCK", "Mengkompilasi Laporan Capaian Bulanan...", "success");
        if (typeof google !== 'undefined' && google.script) {

                    else showNotify("Gagal Cetak", "Data PCK belum lengkap.", "error");
                })
                .getDetailPrintPCK(id);
        }
    }
}

function renderPrintTemplatePK(data) {
    try {
        var container = document.getElementById("template-pk-container");
        if (!container) return;
        document.body.appendChild(container);
        
        var pegawaiNama = state.user.nama;
        var pegawaiJabatan = state.user.jabatan;
        var penilaiNama = "-", penilaiNip = "-";
        var atasanPejabatNama = "-", atasanPejabatNip = "-";
        
        if (state.usersList) {
            state.usersList.forEach(function(u) {
                if (u.nip === data.header.penilai) { penilaiNama = u.nama; penilaiNip = u.nip; }
                if (u.nip === data.header.atasan_pejabat) { atasanPejabatNama = u.nama; atasanPejabatNip = u.nip; }
            });
        }

        var elJabatan = document.getElementById("pk-print-jabatan");
        if(elJabatan) elJabatan.innerText = pegawaiJabatan.toUpperCase();
        
        var tbody = document.getElementById("pk-print-tbody");
        var html = "";
        
        var groups = {};
        data.items.forEach(function(item) {
            if(!groups[item.sasaran]) groups[item.sasaran] = [];
            groups[item.sasaran].push(item);
        });

        var numSasaran = 1;
        for (var sasaran in groups) {
            var kgiatans = groups[sasaran];
            var rowCount = kgiatans.length;
            
            for (var idx = 0; idx < rowCount; idx++) {
                var kg = kgiatans[idx];
                var wktArr = kg.waktu ? kg.waktu.split(",") : [];
                
                html += "<tr>";
                if (idx === 0) {
                    html += "<td rowspan='" + rowCount + "' class='text-center'>" + numSasaran + "</td>";
                    html += "<td rowspan='" + rowCount + "'>" + sasaran + "</td>";
                }
                
                html += "<td class='text-center'>" + (idx + 1) + "</td>";
                html += "<td>" + kg.indikator + "</td>";
                html += "<td class='text-center'>" + kg.mutu + "</td>";
                html += "<td class='text-center'>" + kg.kuantitas + "</td>";
                html += "<td class='text-center'>" + kg.satuan + "</td>";
                
                for (var bln = 1; bln <= 12; bln++) {
                    var cek = wktArr.includes(String(bln)) ? "√" : "";
                    html += "<td class='text-center font-bold'>" + cek + "</td>";
                }
                
                var paguFmt = "-";
                if (kg.pagu && kg.pagu !== "0") {
                    paguFmt = parseInt(kg.pagu).toLocaleString('id-ID');
                }
                html += "<td>IDR <span style='float:right'>" + paguFmt + "</span></td>";
                html += "</tr>";
            }
            numSasaran++;
        }
        
        if(tbody) tbody.innerHTML = html;
        
        var elAtasanNama = document.getElementById("pk-print-atasan-nama");
        var elAtasanNip = document.getElementById("pk-print-atasan-nip");
        var elTgl = document.getElementById("pk-print-tanggal");
        var elPenilaiNama = document.getElementById("pk-print-penilai-nama");
        var elPenilaiNip = document.getElementById("pk-print-penilai-nip");
        
        if(elAtasanNama) elAtasanNama.innerText = atasanPejabatNama;
        if(elAtasanNip) elAtasanNip.innerText = "NIP. " + atasanPejabatNip;
        if(elTgl) elTgl.innerText = "Indramayu, 2 Januari " + data.header.tahun; 
        if(elPenilaiNama) elPenilaiNama.innerText = penilaiNama;
        if(elPenilaiNip) elPenilaiNip.innerText = "NIP. " + penilaiNip;

        document.body.classList.add("print-mode");
        setTimeout(function() { window.print(); document.body.classList.remove("print-mode"); }, 800);
    } catch (e) { console.error("Error di renderPrintTemplatePK:", e); }
}

function toggleBulan(btn) {
    var container = btn.previousElementSibling;
    var checkboxes = container.querySelectorAll('.chk-bulan');
    var allChecked = Array.from(checkboxes).every(function(cb) { return cb.checked; });
    checkboxes.forEach(function(cb) { cb.checked = !allChecked; });
}

function removeIndikatorPK(btn, tbodyId) {
    var tbody = document.getElementById(tbodyId);
    var tr = btn.closest("tr");
    var isFirst = (tr === tbody.querySelector("tr:first-child"));
    
    if (isFirst) {
        showCustomConfirm("Hapus Sasaran Utama", "Menghapus baris ini akan membuang seluruh Indikator di dalam Sasaran ini. Lanjutkan?", function() {
            tbody.remove();
            updateRowNumbersPK();
        });
    } else {
        tr.remove();
        var firstRow = tbody.querySelector("tr:first-child");
        var rowCount = tbody.querySelectorAll("tr").length;
        if (firstRow) {
            firstRow.querySelector(".td-no").rowSpan = rowCount;
            firstRow.querySelector(".td-sasaran").rowSpan = rowCount;
        }
        updateRowNumbersPK();
    }
}

function updateRowNumbersPK() {
    var table = document.getElementById("tablePKItems");

    for (var i = 0; i < tbodies.length; i++) {
        var tdNo = tbodies[i].querySelector("tr:first-child .td-no");
        if (tdNo) tdNo.innerText = (i + 1);
    }
    sasaranCountPK = tbodies.length;
}

                finalStatus = "Menunggu Atasan Penilai";
            } else {
                finalStatus = "Disetujui";
            }
        }
        
        if(btn) { btn.innerHTML = "<i class='fa-solid fa-spinner fa-spin mr-2'></i> Menyimpan..."; btn.disabled = true; }

        var payload = { nipPegawai: nipPegawai, nipPenilai: inputNipPenilai, nipAtasanPejabat: inputNipAtasanPejabat, tahun: inputTahun, items: arrayData, status: finalStatus };

        if (typeof google !== 'undefined' && google.script) {
            google.script.run
                .withSuccessHandler(function(res) {
                    if(btn) { 
                        btn.innerHTML = window.isVerifikasiModePK ? "<i class='fa-solid fa-check-double'></i> Koreksi & Setujui PK" : "<i class='fa-solid fa-floppy-disk'></i> Simpan Seluruh Data PK"; 
                        btn.disabled = false; 
                    }
                    if (res && res.success) {
                        var existingIdx = state.pkpDokumen.findIndex(function(p) { return String(p.id_pkp) === String(res.id_pkp); });
                        if (existingIdx > -1) {
                            state.pkpDokumen[existingIdx].status = finalStatus;
                        } else {
                            state.pkpDokumen.unshift({ id_pkp: res.id_pkp, nip: nipPegawai, penilai: inputNipPenilai, tahun: inputTahun, status: finalStatus, nilai: "0" });
                        }
                        if(typeof renderAllTabelPKP === 'function') { renderAllTabelPKP(); updateBadgeVerifikasiPKP(); }
                        closeModalPK(); 
                        showNotify("Tersimpan", res.message, "success");
                    } else { showNotify("Gagal", res ? res.message : "Error server", "error"); }
                })
                .withFailureHandler(function(error) {
                    if(btn) { btn.disabled = false; }
                    showNotify("Error Koneksi", error.message, "error");
                })

function calcPCKRow(inputEl, kualTarget, groupId) {
    try {
        var tr = inputEl.closest("tr");
        var cellHitung = tr.querySelector(".cell-hitung");
        var cellCapaian = tr.querySelector(".cell-capaian");
        
        var kualReal = parseFloat(inputEl.value);
        if (isNaN(kualReal)) {
            cellHitung.innerText = "-";
            cellCapaian.innerText = "-"; cellCapaian.className = "cell-capaian px-3 py-1 rounded-lg text-slate-600 font-bold";
            recalcPCKGroup(groupId); return;
        }

        var hitung = kualTarget + kualReal;
        var capaian = hitung / 2;

        cellHitung.innerText = hitung;
        cellHitung.className = "p-3 text-center font-black text-slate-200 cell-hitung border-r border-slate-800/50";
        cellCapaian.innerText = capaian.toFixed(2); 

        var colorClass = "bg-emerald-500/10 text-emerald-400 border border-emerald-500/30"; 
        if (capaian < 50) colorClass = "bg-rose-500/10 text-rose-400 border border-rose-500/30";
        else if (capaian <= 75) colorClass = "bg-amber-500/10 text-amber-400 border border-amber-500/30";

        cellCapaian.className = "cell-capaian px-3 py-1.5 rounded-lg font-black " + colorClass;
        recalcPCKGroup(groupId);
    } catch (e) { console.error(e); }
}

function recalcPCKGroup(groupId) {
    try {
        var rows = document.querySelectorAll(".group-" + groupId);
        var totalCap = 0; var totalHit = 0; var count = 0;
        
        for(var i = 0; i < rows.length; i++) {
            var cellCap = rows[i].querySelector(".cell-capaian");
            var cellHit = rows[i].querySelector(".cell-hitung");
            var valCap = parseFloat(cellCap.innerText);
            var valHit = parseFloat(cellHit.innerText);
            if(!isNaN(valCap)) { totalCap += valCap; count++; }
            if(!isNaN(valHit)) { totalHit += valHit; }
        }
        
        var subtotalHitungEl = document.getElementById("subtotal-hitung-group-" + groupId);
        var subtotalEl = document.getElementById("subtotal-group-" + groupId);
        var recapHitungEl = document.getElementById("recap-hitung-" + groupId);
        var recapGroupEl = document.getElementById("recap-group-" + groupId);
        
        if (count > 0) {
            var avgCap = totalCap / count;
            var avgHit = totalHit / count;
            
            if(subtotalHitungEl) subtotalHitungEl.innerText = avgHit.toFixed(2);
            if(subtotalEl) { subtotalEl.innerText = avgCap.toFixed(2); subtotalEl.className = "text-sm font-black text-emerald-400"; }
            
            if(recapHitungEl) recapHitungEl.innerText = avgHit.toFixed(2);
            if(recapGroupEl) recapGroupEl.innerText = avgCap.toFixed(2);
        } else {
            if(subtotalHitungEl) subtotalHitungEl.innerText = "-";
            if(subtotalEl) { subtotalEl.innerText = "-"; subtotalEl.className = "text-sm font-black text-slate-500"; }
            if(recapHitungEl) recapHitungEl.innerText = "-";
            if(recapGroupEl) recapGroupEl.innerText = "-";
        }
        recalcPCKFinal();
    } catch(e) { console.error(e); }
}

function recalcPCKFinal() {
    try {
        var recapsHitung = document.querySelectorAll("[id^='recap-hitung-']");
        var recapsCap = document.querySelectorAll("[id^='recap-group-']");
        
        var totalHit = 0; var countHit = 0;
        var totalCap = 0; var countCap = 0;
        
        for(var j = 0; j < recapsHitung.length; j++) {
            var valHit = parseFloat(recapsHitung[j].innerText);
            if(!isNaN(valHit)) { totalHit += valHit; countHit++; }
        }
        
        for(var i = 0; i < recapsCap.length; i++) {
            var valCap = parseFloat(recapsCap[i].innerText);
            if(!isNaN(valCap)) { totalCap += valCap; countCap++; }
        }
        
        var finalHitungEl = document.getElementById("recap-final-hitung");
        if(finalHitungEl) {
            finalHitungEl.innerText = countHit > 0 ? (totalHit / countHit).toFixed(2) : "-";
        }

        var finalEl = document.getElementById("recap-final-total");
        if(finalEl) {
            if (countCap > 0) {
                var finalAvg = totalCap / countCap;
                var predikat = "<span class='px-2 py-1 bg-rose-500 text-white rounded text-[10px]'>(Kurang)</span>";
                if(finalAvg >= 91) predikat = "<span class='px-2 py-1 bg-emerald-500 text-white rounded text-[10px]'>(Sangat Baik)</span>";
                else if(finalAvg >= 76) predikat = "<span class='px-2 py-1 bg-amber-500 text-white rounded text-[10px]'>(Baik)</span>";
                else if(finalAvg >= 61) predikat = "<span class='px-2 py-1 bg-amber-500 text-white rounded text-[10px]'>(Cukup)</span>";
                else if(finalAvg >= 51) predikat = "<span class='px-2 py-1 bg-amber-500 text-white rounded text-[10px]'>(Kurang)</span>";
                else if(finalAvg <= 50) predikat = "<span class='px-2 py-1 bg-amber-500 text-white rounded text-[10px]'>(Buruk)</span>";


                finalEl.innerHTML = finalAvg.toFixed(2) + predikat;
            } else { finalEl.innerText = "-"; }
        }
    } catch(e) { console.error(e); }
}

function savePCKFull() {
    try {
        var thn = document.getElementById("pckTahunPilih").value;
        var blnSelect = document.getElementById("pckBulanPilih");
        var blnText = blnSelect.options[blnSelect.selectedIndex].text;
        var blnVal = blnSelect.value;

        if(!thn || blnVal === "0") { showNotify("Data Tidak Valid", "Silakan pilih Tahun dan Bulan.", "warning"); return; }

        var rows = document.querySelectorAll(".pck-item-row");
        if(rows.length === 0) { showNotify("Tabel Kosong", "Tidak ada target untuk dilaporkan.", "warning"); return; }

        var items = [];
        var isValid = true;

        for(var i=0; i<rows.length; i++) {
            var idItem = rows[i].getAttribute("data-id");
            var realKuant = rows[i].querySelector(".row-real-kuant").value;
            var realKual = rows[i].querySelector(".row-real-kual").value;
            var capaian = rows[i].querySelector(".cell-capaian").innerText;

            if(realKuant === "" || realKual === "") { isValid = false; break; }

            items.push({ id_item_pk: idItem, real_kuantitas: realKuant, real_mutu: realKual, capaian: capaian });
        }

        if(!isValid) { showNotify("Laporan Belum Lengkap", "Kolom Realisasi belum diisi.", "error"); return; }

        var finalEl = document.getElementById("recap-final-total");
        var nilaiAkhir = finalEl && finalEl.innerText !== "-" ? finalEl.innerText.split("(")[0].trim() : "0";
        
        var nipTarget = window.isVerifikasiModePCK ? window.verifikasiNipTargetPCK : state.user.nip;
        var finalStatus = window.isVerifikasiModePCK ? "Disetujui" : "Draft";

        var payload = { nipPegawai: nipTarget, bulan: blnText, tahun: thn, nilai_akhir: nilaiAkhir, items: items, status: finalStatus };
        var btn = document.getElementById("btnSavePCKFull");
        
        if(btn) { btn.innerHTML = "<i class='fa-solid fa-spinner fa-spin mr-2'></i> Memproses..."; btn.disabled = true; }

        if (typeof google !== 'undefined' && google.script) {
            google.script.run
                .withSuccessHandler(function(res) {
                    if(btn) { btn.disabled = false; }
                    
                    if(res.success) {
                        var existingIdx = state.pckDokumen.findIndex(function(p) { return String(p.id_pck) === String(res.id_pck); });
                        if (existingIdx > -1) {
                            state.pckDokumen[existingIdx].rencana = items.length;
                            state.pckDokumen[existingIdx].nilai_akhir = nilaiAkhir;
                            state.pckDokumen[existingIdx].status = finalStatus;
                        } else {
                            state.pckDokumen.unshift({ id_pck: res.id_pck, nip: nipTarget, bulan: blnText, tahun: thn, rencana: items.length, progress: "100%", status: finalStatus, nilai_akhir: nilaiAkhir });
                        }
                        
                        if(typeof renderAllTabelPKP === 'function') { renderAllTabelPKP(); updateBadgeVerifikasiPKP(); }
                        closeModalPCK();
                        showNotify("Tersimpan", "Dokumen telah diamankan.", "success");
                    } else { showNotify("Gagal", res.message, "error"); }
                })
                .withFailureHandler(function(err) {
                    if(btn) { btn.disabled = false; }
                    showNotify("Error Server", err.message, "error");
                })
                .submitBulkPCK(payload);
        }
    } catch(e) { console.error(e); }
}

function renderAllTabelPKP() {

function renderListPK() {
    try {
        var tbody = document.getElementById("tableBodyListPK");
        if (!tbody) return;
        
        var data = state.pkpDokumen || [];
        var filterTahun = document.getElementById("filterTahunPCK");
        var selectedYear = filterTahun && filterTahun.value ? filterTahun.value : new Date().getFullYear();
        
        var cleanUserNip = String(state.user.nip).replace(/'/g, "").trim();
        
        var myData = data.filter(function(item) { 
            return String(item.nip).replace(/'/g, "").trim() === cleanUserNip && String(item.tahun).trim() === String(selectedYear).trim(); 
        });
        
        if (myData.length === 0) {
            tbody.innerHTML = "<tr><td colspan='5' class='p-8 text-center text-slate-500'>Tidak ada Dokumen PK yang ditemukan pada tahun " + selectedYear + ".</td></tr>";
            return;
        }
        
        var htmlString = "";
        
        for (var i = 0; i < myData.length; i++) {
            var row = myData[i];
            var namaAtasan = row.penilai; 
            
            if (state.usersList) {
                for (var j = 0; j < state.usersList.length; j++) {
                    if (String(state.usersList[j].nip).replace(/'/g, "").trim() === String(row.penilai).replace(/'/g, "").trim()) {
                        namaAtasan = "<span class='font-bold text-slate-200'>" + state.usersList[j].nama + "</span><br><span class='text-[10px] text-slate-500'>" + state.usersList[j].jabatan + "</span>";
                        break;
                    }
                }
            }
            
            var badgeStatus = "bg-slate-500/20 text-slate-400 border border-slate-500/30";
            if(row.status === "Draft") badgeStatus = "bg-amber-500/20 text-amber-400 border border-amber-500/30";
            else if(row.status === "Menunggu Penilai" || row.status === "Menunggu Atasan Penilai") badgeStatus = "bg-blue-500/20 text-blue-400 border border-blue-500/30";
            else if(row.status === "Disetujui" || row.status === "Final") badgeStatus = "bg-emerald-500/20 text-emerald-400 border border-emerald-500/30";
            
            var badgeNilai = "bg-slate-800 text-slate-500";
            if(row.nilai !== "0" && row.nilai !== "-" && row.nilai) badgeNilai = "bg-indigo-500/20 text-indigo-400 font-bold border border-indigo-500/30";
            
            var actionButtons = "";
            if (row.status === "Draft") {
                actionButtons = "<div class='flex items-center justify-center gap-1.5'>" +
                    "<button onclick=\"bukaDetailPK('" + row.id_pkp + "')\" title='Edit Dokumen PK' class='w-8 h-8 rounded-lg bg-slate-800 hover:bg-amber-500 text-slate-300 hover:text-amber-950 shadow transition flex items-center justify-center'><i class='fa-solid fa-pen-to-square'></i></button>" +
                    "<button onclick=\"hapusDokumenPK('" + row.id_pkp + "')\" title='Hapus Dokumen PK' class='w-8 h-8 rounded-lg bg-slate-800 hover:bg-rose-500 text-slate-300 hover:text-white shadow transition flex items-center justify-center'><i class='fa-solid fa-trash'></i></button>" +
                    "<button onclick=\"kirimDokumenPK('" + row.id_pkp + "')\" title='Kirim ke Atasan' class='w-8 h-8 rounded-lg bg-slate-800 hover:bg-indigo-500 text-slate-300 hover:text-white shadow transition flex items-center justify-center'><i class='fa-solid fa-paper-plane'></i></button>" +
                    "<button onclick=\"cetakDokumen('PK', '" + row.id_pkp + "')\" title='Cetak Draft PK' class='w-8 h-8 rounded-lg bg-slate-800 hover:bg-emerald-500 text-slate-300 hover:text-white shadow transition flex items-center justify-center'><i class='fa-solid fa-print'></i></button>" +
                    "</div>";
            } else {
                actionButtons = "<div class='flex items-center justify-center gap-1.5'>" +
                    "<button onclick=\"bukaDetailPK('" + row.id_pkp + "', false, null, true)\" title='Lihat Detail' class='w-8 h-8 rounded-lg bg-slate-800 hover:bg-blue-500 text-slate-300 hover:text-white shadow transition flex items-center justify-center'><i class='fa-solid fa-eye'></i></button>" +
                    "<button onclick=\"cetakDokumen('PK', '" + row.id_pkp + "')\" title='Cetak Final PK' class='w-8 h-8 rounded-lg bg-slate-800 hover:bg-emerald-500 text-slate-300 hover:text-white shadow transition flex items-center justify-center'><i class='fa-solid fa-print'></i></button>" +
                    "</div>";
            }

            htmlString += "<tr class='hover:bg-slate-800/30 transition-colors border-b border-slate-800/50'>" +
                "<td class='p-4 font-bold text-white'><i class='fa-regular fa-calendar text-indigo-400 mr-2'></i> Tahun " + row.tahun + "</td>" +
                "<td class='p-4'>" + namaAtasan + "</td>" +
                "<td class='p-4'><span class='px-2.5 py-1 text-[10px] uppercase tracking-wider font-bold rounded-lg " + badgeStatus + "'>" + row.status + "</span></td>" +
                "<td class='p-4 text-center'><span class='px-3 py-1 text-xs rounded-lg " + badgeNilai + "'>" + (row.nilai === "0" ? "-" : row.nilai) + "</span></td>" +
                "<td class='p-4 text-center'>" + actionButtons + "</td>" +
            "</tr>";
        }
        tbody.innerHTML = htmlString;
    } catch (e) { console.error("Error di renderListPK:", e); }
}

function renderListPCK() {
    try {
        var tbody = document.getElementById("tableBodyListPCK");
        if (!tbody) return;
        
        var filterTahun = document.getElementById("filterTahunPCK");
        var selectedYear = filterTahun && filterTahun.value ? filterTahun.value : new Date().getFullYear();
        var cleanUserNip = String(state.user.nip).replace(/'/g, "").trim();
        
        var dataPCK = (state.pckDokumen || []).filter(function(item) {
            return String(item.tahun).trim() === String(selectedYear).trim() && String(item.nip).replace(/'/g, "").trim() === cleanUserNip;
        });
        
        if (dataPCK.length === 0) {
            tbody.innerHTML = "<tr><td colspan='5' class='p-8 text-center text-slate-500'>Belum ada laporan PCK pada tahun " + selectedYear + ".</td></tr>";
            return;
        }
        
        var htmlString = "";
        for (var i = 0; i < dataPCK.length; i++) {
            var row = dataPCK[i];
            var statusBadge = "";
            var actionButtons = "";
            var nilaiTampil = "-"; 
            
            if (row.status === "Draft") {
                statusBadge = "bg-amber-500/20 text-amber-400 border border-amber-500/30";
                nilaiTampil = "<span class='text-[10px] uppercase tracking-wider text-slate-500'>Draft</span>";
                actionButtons = "<div class='flex items-center justify-center gap-1.5'>" +

                    "<button onclick=\"cetakDokumen('PCK', '" + row.id_pck + "')\" title='Cetak Draft' class='w-8 h-8 rounded-lg bg-slate-800 hover:bg-emerald-500 text-slate-300 hover:text-white shadow transition flex items-center justify-center'><i class='fa-solid fa-print'></i></button>" +

                statusBadge = "bg-indigo-500/20 text-indigo-400 border border-indigo-500/30";
                nilaiTampil = "<span class='text-[10px] uppercase tracking-wider text-indigo-400'>Menunggu...</span>";

            } else if (row.status === "Disetujui" || row.status === "Final") {
                statusBadge = "bg-emerald-500/20 text-emerald-400 border border-emerald-500/30";
                nilaiTampil = row.nilai_akhir ? "<span class='font-black text-emerald-400 text-sm'>" + parseFloat(row.nilai_akhir).toFixed(2) + "</span>" : "-";
                actionButtons = "<div class='flex items-center justify-center gap-1.5'>" +

            }

                "<td class='p-4 font-bold text-white'><i class='fa-regular fa-clock text-amber-500 mr-2'></i> " + row.bulan + " " + selectedYear + "</td>" +
                "<td class='p-4 text-center font-mono text-slate-300'>" + row.rencana + " Kegiatan</td>" +
                "<td class='p-4 text-center'>" + nilaiTampil + "</td>" +
                "<td class='p-4 text-center'><span class='px-2.5 py-1 text-[10px] uppercase tracking-wider font-bold rounded-lg " + statusBadge + "'>" + row.status + "</span></td>" +
                "<td class='p-4 text-center'>" + actionButtons + "</td>" +
            "</tr>";
        }
        tbody.innerHTML = htmlString;
    } catch (e) { console.error("Error di renderListPCK:", e); }
}

function updateBadgeVerifikasiPKP() {
    var count = 0;
    var cleanUserNip = String(state.user.nip).replace(/'/g, "").trim();
    
    if (state.pkpDokumen) {
        count += state.pkpDokumen.filter(function(p) { 
            var isPenilai = (String(p.penilai || "").replace(/'/g, "").trim() === cleanUserNip && p.status === "Menunggu Penilai");
            var isAtasan = (String(p.atasan_pejabat || "").replace(/'/g, "").trim() === cleanUserNip && p.status === "Menunggu Atasan Penilai");
            return isPenilai || isAtasan; 
        }).length;
    }
    
    if (state.pckDokumen && state.pkpDokumen) {
        var bawahanMap = {};
        for (var j = 0; j < state.pkpDokumen.length; j++) {
            bawahanMap[state.pkpDokumen[j].nip + "-" + state.pkpDokumen[j].tahun] = String(state.pkpDokumen[j].penilai).replace(/'/g, "").trim();
        }
        count += state.pckDokumen.filter(function(p) { 
            return p.status === "Menunggu Penilai" && bawahanMap[p.nip + "-" + p.tahun] === cleanUserNip; 
        }).length;
    }
    
    var badge = document.getElementById("badge-verifikasi-pkp");
    var badgeSidebar = document.getElementById("badge-pkp-nav");
    
    if (count > 0) { 
        if (badge) { badge.innerText = count; badge.classList.remove("hidden"); }
        if (badgeSidebar) { badgeSidebar.innerText = count; badgeSidebar.classList.remove("hidden"); }
    } else { 
        if (badge) { badge.classList.add("hidden"); }
        if (badgeSidebar) { badgeSidebar.classList.add("hidden"); }
    }
}

function renderRekapPCK() {
    try {
        var tbody = document.getElementById("tableBodyRekapPCK");
        if (!tbody) return;
        
        var filterTahun = document.getElementById("filterTahunPCK");
        var selectedYear = filterTahun && filterTahun.value ? filterTahun.value : new Date().getFullYear();
        
        // ZETTBOT FIX: Tambahan Filter Bulan
        var filterBulan = document.getElementById("filterBulanPCK");
        var selectedMonth = filterBulan && filterBulan.value !== "0" ? filterBulan.value : null;
        
        var filteredData = (state.pckDokumen || []).filter(function(item) {
            var isTahunSama = String(item.tahun).trim() === String(selectedYear).trim();
            var isBulanSama = selectedMonth ? (String(item.bulan).trim().toLowerCase() === String(selectedMonth).trim().toLowerCase()) : true;

        });

function generateKodePKP(sheetName, prefix) {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(sheetName);
  const lastRow = sheet.getLastRow();
  
  if (lastRow <= 1) return prefix + "-0001";
  
  const lastId = sheet.getRange(lastRow, 1).getDisplayValue();
  const parts = lastId.split("-");
  
  if (parts.length === 2) {
    const num = parseInt(parts[1], 10);
    if (!isNaN(num)) {
      const paddedNum = ("0000" + (num + 1)).slice(-4);
      return prefix + "-" + paddedNum;
    }
  }
  return prefix + "-" + new Date().getTime();
}

function getDataDokumenPKP(nipPegawai, userRole) {
  try {
    const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("pkp_dokumen");
    const lastRow = sheet.getLastRow();
    
    if (lastRow <= 1) return [];
    
    const data = sheet.getRange(2, 1, lastRow - 1, sheet.getLastColumn()).getDisplayValues();
    let result = [];
    const isAdmin = (userRole === "Admin" || userRole === "Operator");
    const cleanNipPegawai = String(nipPegawai).replace(/'/g, "").trim();
    
    for (let i = 0; i < data.length; i++) {
      let dbNip = String(data[i][1]).replace(/'/g, "").trim();
      let dbPenilai = String(data[i][2]).replace(/'/g, "").trim();
      // ZETTBOT FIX: Sertakan Atasan Pejabat Penilai di Data Puller agar Frontend bisa memfilter Hak Akses
      let dbAtasan = String(data[i][7] || "").replace(/'/g, "").trim(); 
      
      if (isAdmin || dbNip === cleanNipPegawai || dbPenilai === cleanNipPegawai || dbAtasan === cleanNipPegawai) { 
        result.push({
          id_pkp: data[i][0],
          nip: data[i][1],
          penilai: data[i][2],
          tahun: data[i][3],
          status: data[i][4],
          nilai: data[i][5],
          atasan_pejabat: data[i][7]
        });
      }
    }
    return result;
  } catch (error) {
    return [];
  }
}

function getItemsPKP(idPKP) {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const sheetRencana = ss.getSheetByName("pkp_rencana");
    const sheetEvaluasi = ss.getSheetByName("pkp_evaluasi");
    
    const lastRowRencana = sheetRencana.getLastRow();
    if (lastRowRencana <= 1) return [];
    const dataRencana = sheetRencana.getRange(2, 1, lastRowRencana - 1, sheetRencana.getLastColumn()).getDisplayValues();
    
    let dataEvaluasi = [];
    if(sheetEvaluasi) {
        const lastRowEval = sheetEvaluasi.getLastRow();
        if (lastRowEval > 1) {
            dataEvaluasi = sheetEvaluasi.getRange(2, 1, lastRowEval - 1, sheetEvaluasi.getLastColumn()).getDisplayValues();
        }
    }
    
    let result = [];
    for (let i = 0; i < dataRencana.length; i++) {
      if (dataRencana[i][1] === idPKP) { 
        let idItem = dataRencana[i][0];
        let totalCapaian = 0;
        
        for(let j = 0; j < dataEvaluasi.length; j++) {
            if(dataEvaluasi[j][1] === idItem) { 
                let angka = parseFloat(dataEvaluasi[j][3]) || 0; 
                totalCapaian += angka;
            }
        }
        
        result.push({
          id_item: idItem,
          id_pkp: dataRencana[i][1],
          jenis: dataRencana[i][2],
          sasaran: dataRencana[i][3], 
          indikator: dataRencana[i][4], 
          target: parseFloat(dataRencana[i][5]) || 0, 
          satuan: dataRencana[i][6],
          capaian: totalCapaian
        });
      }
    }
    return result;
  } catch (error) {
    return []; 
  }
}

function getPkpDb() {
  return SpreadsheetApp.openById(PropertiesService.getScriptProperties().getProperty("TRANSKEP_DB_ID")) || SpreadsheetApp.getActiveSpreadsheet();
}

function checkAndCreateSheetPKP(sheetName, headers) {
  if (!sheetName || !headers || headers.length === 0) return null;

  var ss = getPkpDb();
  var sheet = ss.getSheetByName(sheetName);
  
  if (!sheet) {
    sheet = ss.insertSheet(sheetName);
    sheet.appendRow(headers);
    sheet.getRange(1, 1, 1, headers.length).setFontWeight("bold").setBackground("#064e3b").setFontColor("#ffffff");
  } else {
    var currentHeaders = sheet.getRange(1, 1, 1, sheet.getLastColumn() || 1).getValues()[0];
    if (sheetName === "pkp_rencana" && currentHeaders.length !== headers.length) {
      sheet.getRange(1, 1, 1, headers.length).setValues([headers]);
      sheet.getRange(1, 1, 1, headers.length).setFontWeight("bold").setBackground("#064e3b").setFontColor("#ffffff");
    }
  }
  return sheet;
}

function preparePKPDatabase() {
  checkAndCreateSheetPKP("pkp_dokumen", ["ID_PKP", "NIP_Pegawai", "NIP_Penilai", "Tahun", "Status", "Nilai_Akhir", "Timestamp", "NIP_Atasan_Pejabat"]);
  checkAndCreateSheetPKP("pkp_rencana", ["ID_Item", "ID_PKP", "Jenis", "Sasaran_Kegiatan", "Indikator_Kinerja", "Kuantitas", "Satuan", "Mutu", "Waktu", "Pagu_Anggaran", "Status_Item", "Timestamp"]);
  checkAndCreateSheetPKP("pck_dokumen", ["ID_PCK", "NIP_Pegawai", "Bulan", "Tahun", "Total_Rencana", "Progress", "Status", "Nilai_Akhir", "Timestamp"]);
  checkAndCreateSheetPKP("pck_realisasi", ["ID_Realisasi", "ID_PCK", "ID_Item_PK", "Real_Kuantitas", "Real_Mutu", "Capaian", "Timestamp"]);
  checkAndCreateSheetPKP("master_kinerja", ["ID_Tugas", "Jabatan", "Kategori", "Sasaran_Kegiatan", "Indikator_Kinerja", "Satuan_Default", "Mutu_Default", "Waktu_Default"]);
}

function submitBulkPKP(payload) {
  var lock = LockService.getScriptLock();
  try {
    lock.waitLock(15000); 
    
    var ss = getPkpDb();
    var sheetDokumen = ss.getSheetByName("pkp_dokumen");
    var sheetRencana = ss.getSheetByName("pkp_rencana");
    var timestamp = Utilities.formatDate(new Date(), 'Asia/Jakarta', 'dd/MM/yyyy HH:mm:ss');
    
    var expectedRencanaHeaders = ["ID_Item", "ID_PKP", "Jenis", "Sasaran_Kegiatan", "Indikator_Kinerja", "Kuantitas", "Satuan", "Mutu", "Waktu", "Pagu_Anggaran", "Status_Item", "Timestamp"];
    var checkHeader = sheetRencana.getRange(1, 1, 1, expectedRencanaHeaders.length).getValues()[0];
    
    if (String(checkHeader[3]).trim() !== "Sasaran_Kegiatan") {
      sheetRencana.getRange(1, 1, 1, expectedRencanaHeaders.length).setValues([expectedRencanaHeaders]);
      sheetRencana.getRange(1, 1, 1, expectedRencanaHeaders.length).setFontWeight("bold").setBackground("#064e3b").setFontColor("#ffffff");
    }
    
    var docData = sheetDokumen.getDataRange().getDisplayValues();
    var currentPkpId = null;
    var docRowIndex = -1;
    
    for (var i = 1; i < docData.length; i++) {
      if (String(docData[i][1]).replace(/'/g, "").trim() === String(payload.nipPegawai).trim() && 
          String(docData[i][3]).trim() === String(payload.tahun).trim()) {
        currentPkpId = docData[i][0];
        docRowIndex = i + 1;
        break;
      }
    }
    
    var finalStatus = payload.status || "Draft";
    
    if (currentPkpId) {
      sheetDokumen.getRange(docRowIndex, 3).setValue(payload.nipPenilai);
      sheetDokumen.getRange(docRowIndex, 8).setValue(payload.nipAtasanPejabat);
      sheetDokumen.getRange(docRowIndex, 5).setValue(finalStatus);
    } else {
      currentPkpId = "PKP-" + new Date().getTime();
      sheetDokumen.appendRow([currentPkpId, payload.nipPegawai, payload.nipPenilai, payload.tahun, finalStatus, "0", timestamp, payload.nipAtasanPejabat]);
    }
    
    var renData = sheetRencana.getDataRange().getDisplayValues();
    var rowsToDelete = [];
    for (var j = renData.length - 1; j >= 1; j--) {
      if (String(renData[j][1]).trim() === String(currentPkpId).trim()) {
        rowsToDelete.push(j + 1);
      }
    }
    rowsToDelete.forEach(function(rowNum) { sheetRencana.deleteRow(rowNum); });
    
    if (payload.items && payload.items.length > 0) {
      var rowsToInsert = [];
      var baseNum = new Date().getTime();
      for (var k = 0; k < payload.items.length; k++) {
        var item = payload.items[k];
        var itemId = "ITEM-" + (baseNum + k);
        rowsToInsert.push([
          itemId, currentPkpId, "Utama", item.sasaran, item.indikator, 
          item.kuantitas, item.satuan, item.mutu, item.waktu, item.pagu, "Aktif", timestamp
        ]);
      }
      if (rowsToInsert.length > 0) {
        sheetRencana.getRange(sheetRencana.getLastRow() + 1, 1, rowsToInsert.length, rowsToInsert[0].length).setValues(rowsToInsert);
      }
    }
    
    SpreadsheetApp.flush();
    return { success: true, id_pkp: currentPkpId, message: "Seluruh data PK berhasil disimpan." };
  } catch (e) {
    return { success: false, message: e.toString() };
  } finally {
    lock.releaseLock(); 
  }
}

function getDetailPKP(idPkp) {
  try {
    var ss = getPkpDb();
    var sheetDokumen = ss.getSheetByName("pkp_dokumen");
    var docData = sheetDokumen.getDataRange().getDisplayValues();
    var header = null;
    var safeIdPkp = String(idPkp).trim();
    
    for (var i = 1; i < docData.length; i++) {
      if (String(docData[i][0]).trim() === safeIdPkp) {
        header = {
          id_pkp: docData[i][0], nip: docData[i][1], penilai: docData[i][2], 
          tahun: docData[i][3], status: docData[i][4], nilai: docData[i][5],
          atasan_pejabat: docData[i][7]
        };
        break;
      }
    }
    if (!header) throw new Error("Dokumen PKP tidak ditemukan.");
    
    var sheetRencana = ss.getSheetByName("pkp_rencana");
    var renData = sheetRencana.getDataRange().getDisplayValues();
    var items = [];
    
    for (var j = 1; j < renData.length; j++) {
      if (String(renData[j][1]).trim() === safeIdPkp) {
        items.push({
          id_item: renData[j][0], 
          sasaran: renData[j][3], 
          indikator: renData[j][4], 
          kuantitas: renData[j][5], 
          satuan: renData[j][6], 
          mutu: renData[j][7],
          waktu: renData[j][8], 
          pagu: renData[j][9] || "0"
        });
      }
    }
    return { success: true, header: header, items: items };
  } catch (e) {
    return { success: false, message: e.toString() };
  }
}

function getDataDokumenPCK(nipPegawai, userRole) {
  try {
    var ss = getPkpDb();
    var sheetPCK = ss.getSheetByName("pck_dokumen");
    var sheetPKP = ss.getSheetByName("pkp_dokumen");
    
    var pckLastRow = sheetPCK.getLastRow();
    if (pckLastRow <= 1) return [];
    
    var pckData = sheetPCK.getRange(2, 1, pckLastRow - 1, sheetPCK.getLastColumn()).getDisplayValues();
    var pkpData = sheetPKP.getDataRange().getDisplayValues();
    
    var bawahanMap = {};
    for (var j = 1; j < pkpData.length; j++) {
        bawahanMap[pkpData[j][1] + "-" + pkpData[j][3]] = pkpData[j][2]; 
    }
    
    var result = [];
    var isAdmin = (userRole === "Admin" || userRole === "Operator");
    var cleanNipPegawai = String(nipPegawai).replace(/'/g, "").trim();
    
    for (var i = 0; i < pckData.length; i++) {
      var nipPemilik = String(pckData[i][1]).replace(/'/g, "").trim();
      var tahunDoc = pckData[i][3];
      var nipPenilaiDoc = String(bawahanMap[nipPemilik + "-" + tahunDoc] || "").replace(/'/g, "").trim();
      
      if (isAdmin || nipPemilik === cleanNipPegawai || nipPenilaiDoc === cleanNipPegawai) { 
        result.push({
          id_pck: pckData[i][0],
          nip: pckData[i][1],
          bulan: pckData[i][2],
          tahun: pckData[i][3],
          rencana: pckData[i][4],
          progress: pckData[i][5],
          status: pckData[i][6],
          nilai_akhir: pckData[i][7]
        });
      }
    }
    return result;
  } catch (e) { return []; }
}

function getDetailLengkapPCK(nipPegawai, bulanStr, tahunStr) {
  try {
    var ss = getPkpDb();
    var sheetPK = ss.getSheetByName("pkp_dokumen");
    var docPK = sheetPK.getDataRange().getDisplayValues();
    var idPkpFound = null;
    
    var cleanNip = String(nipPegawai).replace(/'/g, "").trim();
    
    for (var i = 1; i < docPK.length; i++) {
      if (String(docPK[i][1]).replace(/'/g, "").trim() === cleanNip && String(docPK[i][3]).trim() === String(tahunStr).trim()) {
        idPkpFound = docPK[i][0];
        break;
      }
    }
    
    if (!idPkpFound) return { success: false, message: "Dokumen PK tahun " + tahunStr + " belum dibuat. Buat PK terlebih dahulu." };
    
    var sheetRencana = ss.getSheetByName("pkp_rencana");
    var renData = sheetRencana.getDataRange().getDisplayValues();
    var targets = [];
    
    for (var j = 1; j < renData.length; j++) {
      if (String(renData[j][1]).trim() === String(idPkpFound).trim()) {
        targets.push({ 
          id_item: renData[j][0], 
          sasaran: renData[j][3], 
          indikator: renData[j][4], 
          kuantitas: renData[j][5], 
          satuan: renData[j][6], 
          mutu: renData[j][7] 
        });
      }
    }

    var sheetPCK = ss.getSheetByName("pck_dokumen");
    var docPCK = sheetPCK.getDataRange().getDisplayValues();
    var idPckFound = null;
    
    for (var k = 1; k < docPCK.length; k++) {
      if (String(docPCK[k][1]).replace(/'/g, "").trim() === cleanNip && 
          String(docPCK[k][2]).trim() === String(bulanStr).trim() && 
          String(docPCK[k][3]).trim() === String(tahunStr).trim()) {
        idPckFound = docPCK[k][0];
        break;
      }
    }

    if (idPckFound) {
      var sheetReal = ss.getSheetByName("pck_realisasi");
      var realData = sheetReal.getDataRange().getDisplayValues();
      for (var t = 0; t < targets.length; t++) {
        for (var r = 1; r < realData.length; r++) {
          if (String(realData[r][1]).trim() === String(idPckFound).trim() && String(realData[r][2]).trim() === String(targets[t].id_item).trim()) {
            targets[t].real_kuantitas = realData[r][3];
            targets[t].real_mutu = realData[r][4];
            targets[t].capaian = realData[r][5];
            break;
          }
        }
      }
    }

    return { success: true, targets: targets };
  } catch(e) {
    return { success: false, message: e.toString() };
  }
}

function aksiDokumenPKP(idPKP, aksi) {

    var sheet = getPkpDb().getSheetByName("pkp_dokumen");

      if (String(data[i][0]) === String(idPKP)) {
        if (aksi === "Hapus") {
            sheet.deleteRow(i + 1);

            if (currStatus === "Menunggu Penilai") {

            } else {
                sheet.getRange(i + 1, 5).setValue("Disetujui");
            }
        }
        SpreadsheetApp.flush();

      }
    }
  } catch(e) { return { success: false }; }

function getDetailPrintPCK(idPCK) {
   try {
    var ss = getPkpDb();
    var sheetPCK = ss.getSheetByName("pck_dokumen");
    var docPCK = sheetPCK.getDataRange().getDisplayValues();
    var header = null;
    
    for (var i = 1; i < docPCK.length; i++) {
      if (String(docPCK[i][0]) === String(idPCK)) {
        header = { id_pck: docPCK[i][0], nip: docPCK[i][1], bulan: docPCK[i][2], tahun: docPCK[i][3] };
        break;
      }
    }
    if(!header) throw new Error("Data PCK tidak ditemukan");
    
    var result = getDetailLengkapPCK(header.nip, header.bulan, header.tahun);
    if(!result.success) throw new Error(result.message);
    
    var sheetPK = ss.getSheetByName("pkp_dokumen");
    var docPK = sheetPK.getDataRange().getDisplayValues();
    var penilai_nip = "-";
    for (var k = 1; k < docPK.length; k++) {
      if (String(docPK[k][1]).trim() === String(header.nip).trim() && String(docPK[k][3]).trim() === String(header.tahun).trim()) {
        penilai_nip = docPK[k][2];
        break;
      }
    }

    return { success: true, header: header, items: result.targets, penilai_nip: penilai_nip };
  } catch(e) { return { success: false, message: e.toString() }; }
}
