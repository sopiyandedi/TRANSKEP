function submitIzinPribadi(payload) {
  try {
    const ss = SpreadsheetApp.openById(PropertiesService.getScriptProperties().getProperty("TRANSKEP_DB_ID"));
    let sheetIP = ss.getSheetByName('izin_pribadi_log');
    
    // ZettBOT Feature: Lazy-Initialization (Auto-Create jika sheet belum ada)
    if(!sheetIP) {
        sheetIP = ss.insertSheet('izin_pribadi_log');
        const headers = ['ID', 'Waktu_Pengajuan', 'NIP', 'Tgl_Mulai', 'Tgl_Selesai', 'Alasan', 'Atasan_NIP', 'Status', 'Catatan_Atasan'];
        sheetIP.appendRow(headers);
        sheetIP.getRange("A1:I1").setFontWeight("bold").setBackground("#064e3b").setFontColor("#ffffff");
        sheetIP.setFrozenRows(1);
    }
    
    // Format Waktu & ID Unik
    const timestamp = Utilities.formatDate(new Date(), 'Asia/Jakarta', 'dd/MM/yyyy HH:mm:ss');
    const newId = "IP-" + new Date().getTime() + Math.floor(Math.random() * 1000);
    
    // Batch Insert 1 baris
    sheetIP.appendRow([
        newId,
        timestamp,
        payload.nip,
        payload.tgl_mulai,
        payload.tgl_selesai,
        payload.alasan,
        payload.atasan_nip,
        'Menunggu',
        ''
    ]);
    
    SpreadsheetApp.flush();
    return { status: 'success', message: 'Izin Pribadi berhasil diajukan dan sedang menunggu persetujuan.', data: { id: newId } };
    
  } catch (err) { 
    return { status: 'error', message: err.toString() }; 
  }
}

function getIzinPribadiById(id) {
  try {
    const ss = SpreadsheetApp.openById(PropertiesService.getScriptProperties().getProperty("TRANSKEP_DB_ID"));
    const sheetIP = ss.getSheetByName('izin_pribadi_log');
    if (!sheetIP) throw new Error("Data Izin Pribadi belum tersedia.");
    
    const data = sheetIP.getDataRange().getDisplayValues();
    for (let i = 1; i < data.length; i++) {
      if (String(data[i][0]) === String(id)) {
        return {
          status: 'success',
          data: {
            id: data[i][0],
            waktu: data[i][1],
            nip: data[i][2],
            tgl_mulai: data[i][3],
            tgl_selesai: data[i][4],
            alasan: data[i][5],
            atasan_nip: data[i][6],
            status: data[i][7],
            catatan: data[i][8]
          }
        };
      }
    }
    throw new Error("Data Izin Pribadi tidak ditemukan.");
  } catch (err) {
    return { status: 'error', message: err.toString() };
  }
}

function updateIpStatus(id, decision, notes, userNip, bypassMode = null) {
  try {
    const ss = SpreadsheetApp.openById(PropertiesService.getScriptProperties().getProperty("TRANSKEP_DB_ID"));
    const sheet = ss.getSheetByName('izin_pribadi_log');
    const data = sheet.getDataRange().getValues();
    
    let rowIndex = -1;
    for (let i = 1; i < data.length; i++) {
      if (String(data[i][0]) === String(id)) { rowIndex = i + 1; break; }
    }
    if (rowIndex === -1) throw new Error("Data Izin Pribadi tidak ditemukan di sistem.");
    
    let finalNotes = notes;
    if (bypassMode) finalNotes += "\n\n*(Disetujui via Bypass Atasan)*";
    
    sheet.getRange(rowIndex, 8).setValue(decision); // Set Kolom Status (H)
    sheet.getRange(rowIndex, 9).setValue(finalNotes); // Set Kolom Catatan (I)
    
    SpreadsheetApp.flush();
    return { status: 'success', message: 'Usulan Izin Pribadi berhasil diproses.' };
  } catch(e) { 
    return { status: 'error', message: e.toString() }; 
  }
}

function deleteIzinPribadiLog(id, forceDelete = false) {
  try {
    const ss = SpreadsheetApp.openById(PropertiesService.getScriptProperties().getProperty("TRANSKEP_DB_ID"));
    const sheet = ss.getSheetByName('izin_pribadi_log');
    const data = sheet.getDataRange().getValues();
    
    let rowIndex = -1;
    for (let i = 1; i < data.length; i++) {
      if (String(data[i][0]) === String(id)) { 
        if (!forceDelete && data[i][7] !== 'Menunggu') throw new Error("Akses Ditolak: Hanya usulan berstatus Menunggu yang dapat ditarik/dihapus.");
        rowIndex = i + 1; 
        break; 
      }
    }
    if (rowIndex === -1) throw new Error("Data tidak ditemukan di sistem.");
    
    sheet.deleteRow(rowIndex);
    SpreadsheetApp.flush();
    return { status: 'success', message: 'Pengajuan Izin Pribadi berhasil ditarik dan dihapus.' };
  } catch (err) { return { status: 'error', message: err.toString() }; }
}

// ZETTBOT FIX: Fungsi Soft Edit / Update Data IP
function updateIzinPribadiLog(payload) {
  try {
    const ss = SpreadsheetApp.openById(PropertiesService.getScriptProperties().getProperty("TRANSKEP_DB_ID"));
    const sheet = ss.getSheetByName('izin_pribadi_log');
    const data = sheet.getDataRange().getValues();
    
    let rowIndex = -1;
    for (let i = 1; i < data.length; i++) {
      if (String(data[i][0]) === String(payload.id)) { rowIndex = i + 1; break; }
    }
    if (rowIndex === -1) throw new Error("Data tidak ditemukan.");
    
    // Cek Keamanan
    if (data[rowIndex-1][7] !== 'Menunggu' && data[rowIndex-1][7] !== 'Ditolak') {
        throw new Error("Akses Ditolak: Hanya data berstatus Menunggu atau Ditolak yang bisa diubah.");
    }
    
    sheet.getRange(rowIndex, 4).setValue(payload.tgl_mulai);
    sheet.getRange(rowIndex, 5).setValue(payload.tgl_selesai);
    sheet.getRange(rowIndex, 6).setValue(payload.alasan);
    sheet.getRange(rowIndex, 7).setValue(payload.atasan_nip);
    sheet.getRange(rowIndex, 8).setValue('Menunggu'); // Reset Status
    sheet.getRange(rowIndex, 9).setValue(''); // Reset Catatan
    
    SpreadsheetApp.flush();
    return { status: 'success', message: 'Izin Pribadi berhasil diperbarui.' };
  } catch (err) { return { status: 'error', message: err.toString() }; }
}

    function cetakIzinPribadi(id) {
        if (window.isGlobalPrinting) {
            showNotify("Proses Sibuk", "Sistem sedang menyiapkan dokumen. Harap tunggu.", "warning");
            return; 
        }

        if (!state || !state.user) {

        showLoading("Menyiapkan dokumen Izin Pribadi...");

                    if (res.status === 'success') {
                        renderTemplateCetakIP(res.data);
                    } else {
                        hideLoading();
                        window.isGlobalPrinting = false;
                        showNotify("Gagal", res.message, "error");
                    }

                    hideLoading();
                    window.isGlobalPrinting = false;
                    showNotify("Error Koneksi", err.toString(), "error");

            setTimeout(() => {
                hideLoading();
                window.isGlobalPrinting = false;
                alert("[CANVAS MODE] Simulasi cetak untuk ID: " + id);
            }, 1000);
        }
    }

<script>
    // 1. FUNGSI MEMBUKA MODAL
    let currentEditIdIP = null; 

    function openModalIP(editId = null) {
        if (!state || !state.user) {
            showNotify("Sistem Memuat", "Silakan tunggu, data sistem belum siap.", "error");
            return;
        }
        
        currentEditIdIP = editId;
        document.getElementById('formIP').reset();
        
        const selAtasan = document.getElementById('ipAtasan');
        const alertPlh = document.getElementById('alertPlhIP');
        selAtasan.innerHTML = '<option value="">-- Pilih Pejabat --</option>';
        
        let pejabatOpts = [];
        let adaPlh = false;
        
        let ketua = state.usersList.find(x => x.jabatan.toLowerCase() === "ketua");
        
        if (ketua && typeof isPegawaiAbsent === 'function' && !isPegawaiAbsent(ketua.nip)) {
            pejabatOpts.push(ketua);
        } else {
            adaPlh = true;
            let wakil = state.usersList.find(x => x.jabatan.toLowerCase() === "wakil ketua");
            if (wakil && typeof isPegawaiAbsent === 'function' && !isPegawaiAbsent(wakil.nip)) {
                pejabatOpts.push(wakil); 
            } else { 
                let hakims = state.usersList.filter(x => x.jabatan.toLowerCase().includes("hakim") && (typeof isPegawaiAbsent === 'function' ? !isPegawaiAbsent(x.nip) : true)); 
                pejabatOpts.push(...hakims); 
            }
        }
        
        if (adaPlh && alertPlh) alertPlh.classList.remove('hidden');
        else if (alertPlh) alertPlh.classList.add('hidden');
        
        pejabatOpts.forEach(o => { 
            selAtasan.innerHTML += `<option value="${o.nip}">${o.nama} (${o.jabatan})</option>`; 
        });
        
        if (pejabatOpts.length === 1) selAtasan.value = pejabatOpts[0].nip;

        if (currentEditIdIP) {
            const log = state.ipLogs.find(l => String(l.id) === String(currentEditIdIP));
            if (log) {
                document.getElementById('ipMulai').value = log.tgl_mulai;
                document.getElementById('ipSelesai').value = log.tgl_selesai;
                document.getElementById('ipAlasan').value = log.alasan;
                document.getElementById('ipAtasan').value = log.atasan_nip;
            }
        }

        const modal = document.getElementById('modalFormIP');
        if (modal.parentNode !== document.body) {
            document.body.appendChild(modal); 
        }
        
        modal.classList.remove('hidden');
        setTimeout(() => { 
            if(modal.firstElementChild) modal.firstElementChild.classList.remove('scale-95'); 
        }, 10);
    }

    // 2. FUNGSI MENUTUP MODAL
    function closeModalIP() {
        const modal = document.getElementById('modalFormIP');
        if(modal && modal.firstElementChild) {
            modal.firstElementChild.classList.add('scale-95');
            setTimeout(() => { modal.classList.add('hidden'); }, 200);
        }
    }

    // 3. FUNGSI SUBMIT KE BACKEND
    function submitFormIP(e) {
        e.preventDefault();
        
        if(!state.user) {
            showNotify("Sesi Berakhir", "Harap login ulang.", "error");
            return;
        }
        
        // ZETTBOT FIX: Validasi Hirarki Saldo Cuti Tahunan
        let totalTahunan = Number(state.user.s_tahunan_2024 || 0) + Number(state.user.s_tahunan_2025 || 0) + Number(state.user.s_tahunan_2026 || 0);
        if (totalTahunan > 0) {
            showNotify("Pengajuan Ditolak", "Izin Pribadi hanya bisa dipakai jika Saldo Cuti Tahunan Anda sudah benar-benar habis. Sisa cuti tahunan Anda: " + totalTahunan + " Hari.", "warning");
            return;
        }

        const payload = {
            id: currentEditIdIP, 
            nip: state.user.nip,
            tgl_mulai: document.getElementById('ipMulai').value,
            tgl_selesai: document.getElementById('ipSelesai').value,
            alasan: document.getElementById('ipAlasan').value,
            atasan_nip: document.getElementById('ipAtasan').value
        };
        
        if (new Date(payload.tgl_selesai) < new Date(payload.tgl_mulai)) {
            showNotify("Tanggal Invalid", "Selesai pelaksanaan tidak boleh sebelum tanggal mulai.", "error");
            return;
        }

        showLoading(currentEditIdIP ? "Memperbarui Izin Pribadi..." : "Mengirimkan Izin Pribadi...");
        
        if (typeof google !== 'undefined') {
            const runner = google.script.run
              .withSuccessHandler(function(res) {
                 hideLoading();
                 if(res.status === 'success') {
                    closeModalIP();
                    showNotify("Berhasil Disimpan", res.message, "success");
                    
                    if (currentEditIdIP) {
                        const idx = state.ipLogs.findIndex(l => String(l.id) === String(currentEditIdIP));
                        if (idx !== -1) {
                            state.ipLogs[idx].tgl_mulai = payload.tgl_mulai;
                            state.ipLogs[idx].tgl_selesai = payload.tgl_selesai;
                            state.ipLogs[idx].alasan = payload.alasan;
                            state.ipLogs[idx].atasan_nip = payload.atasan_nip;
                            state.ipLogs[idx].status = 'Menunggu';
                        }
                    } else {
                        const virtualLog = {
                            id: (res.data && res.data.id) ? res.data.id : "IP-VIRTUAL-" + Date.now(),
                            tgl_pengajuan: new Date().toLocaleDateString('id-ID'),
                            nip: payload.nip,
                            tgl_mulai: payload.tgl_mulai,
                            tgl_selesai: payload.tgl_selesai,
                            alasan: payload.alasan,
                            atasan_nip: payload.atasan_nip,
                            status: 'Menunggu',
                            catatan: '',
                            z_type: 'IP',
                            z_sort: Date.now()
                        };
                        if (state && state.ipLogs) {
                            state.ipLogs.unshift(virtualLog);
                        }
                    }
                    
                    if (typeof renderPermohonanView === 'function') {
                        renderPermohonanView();
                    }
                 } else {
                    showNotify("Gagal", res.message, "error");
                 }
              })
              .withFailureHandler(function(err) {
                 hideLoading();
                 showNotify("Error Jaringan", err.toString(), "error");
              });
              
            if (currentEditIdIP) {
                runner.updateIzinPribadiLog(payload);
            } else {
                runner.submitIzinPribadi(payload);
            }
        } else {
            setTimeout(() => {
                hideLoading();
                closeModalIP();
                alert("[CANVAS MODE] Simulasi berhasil! Izin Pribadi diproses.\n" + JSON.stringify(payload));
            }, 1000);
        }
    }
</script>
